    67→    67→function ValidateStep() {
    68→    68→  const { currentImportId, setLoading, setError, setValidationResult, setStep, isLoading } = useETLStore();
    69→    69→  const handleProceed = useCallback(() => { setStep(3); }, [setStep]);
    70→    70→
    71→    71→  useEffect(() => {
    72→    72→    const runValidation = async () => {
    73→    73→      if (!currentImportId) return;
    74→    74→      setLoading(true);
    75→    75→      setError(null);
    76→    76→      try {
    77→    77→        const res = await fetch(`/api/imports/${currentImportId}/validate`, { method: 'POST' });
    78→    78→        if (!res.ok) { const err = await res.json().catch(() => ({})); throw new Error(err.error || 'Validation failed'); }
    79→    79→        const json = await res.json();
    80→    80→        if (!json.success) throw new Error(json.error || 'Validation failed');
    81→    81→        setValidationResult(json.data as ValidationResultClient);
    82→    82→      } catch (err) { setError(err instanceof Error ? err.message : 'Validation failed'); }
    83→    83→      finally { setLoading(false); }
    84→    84→    };
    85→    85→    runValidation();
    86→    86→  }, [currentImportId, setLoading, setError, setValidationResult]);
    87→    87→
    88→    88→  return (
    89→    89→    <div>
    90→    90→      {isLoading && (
    91→    91→        <Card className="mb-4">
    92→    92→          <CardContent className="py-8 text-center">
    93→    93→            <Loader2 className="h-8 w-8 mx-auto text-blue-500 animate-spin mb-3" />
    94→    94→            <p className="text-sm text-gray-600">Running 3-phase validation...</p>
    95→    95→            <p className="text-xs text-gray-400 mt-1">Schema → Data → Business Rules</p>
    96→    96→          </CardContent>
    97→    97→        </Card>
    98→    98→      )}
    99→    99→      <ValidationResults onProceed={handleProceed} />
   100→   100→    </div>
   101→   101→  );
   102→   102→}

   104→   104→function HarmonizeStep() {
   105→   105→  const { currentImportId, setLoading, setError, setHarmonizationResult, setStep, harmonizationResult, isLoading } = useETLStore();
   106→   106→  const handleProceed = useCallback(() => { setStep(4); }, [setStep]);
   107→   107→
   108→   108→  useEffect(() => {
   109→   109→    if (harmonizationResult) return;
   110→   110→    const runHarmonization = async () => {
   111→   111→      if (!currentImportId) return;
   112→   112→      setLoading(true); setError(null);
   113→   113→      try {
   114→   114→        const res = await fetch(`/api/imports/${currentImportId}/harmonize`, { method: 'POST' });
   115→   115→        if (!res.ok) { const err = await res.json().catch(() => ({})); throw new Error(err.error || 'Harmonization failed'); }
   116→   116→        const json = await res.json();
   117→   117→        if (!json.success) throw new Error(json.error || 'Harmonization failed');
   118→   118→        setHarmonizationResult(json.data as HarmonizationResultClient);
   119→   119→      } catch (err) { setError(err instanceof Error ? err.message : 'Harmonization failed'); }
   120→   120→      finally { setLoading(false); }
   121→   121→    };
   122→   122→    runHarmonization();
   123→   123→  }, [currentImportId, harmonizationResult, setLoading, setError, setHarmonizationResult]);
   124→   124→
   125→   125→  if (isLoading && !harmonizationResult) {
   126→   126→    return (
   127→   127→      <Card>
