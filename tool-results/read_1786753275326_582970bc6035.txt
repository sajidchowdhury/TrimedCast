    59→
    60→export function calculateTotalLeadTime(config: LeadTimeConfig = DEFAULT_LEAD_TIME): {
    61→  total: number;
    62→  breakdown: { manufacturing: number; shipping: number; customs: number; internal: number };
    63→  variance: number;
    64→} {
    65→  const shipping = config.shippingMethod === 'sea' ? config.seaTransitDays : config.airTransitDays;
    66→  const customs = config.shippingMethod === 'sea' ? config.customsClearanceDays : 3;
    67→
    68→  return {
    69→    total: config.manufacturingDays + shipping + customs + config.internalProcessingDays,
    70→    breakdown: {
    71→      manufacturing: config.manufacturingDays,
    72→      shipping,
    73→      customs,
    74→      internal: config.internalProcessingDays,
    75→    },
    76→    variance: Math.sqrt(
    77→      Math.pow(config.manufacturingVariance, 2) +
    78→      Math.pow(config.shippingVariance, 2) +
    79→      Math.pow(config.customsVariance, 2)
    80→    ),
    81→  };
    82→}

    88→export interface LeadTimeDistribution {

    90→  mfgBest: number;       // Best case (75 days)

    99→  // Shipment (Air)

   102→  shipAirWorst: number;  // 12 days
   103→
   104→  // Customs (Sea/Chittagong)
   105→  customsSeaBest: number;    // 7 days
   106→  customsSeaLikely: number;  // 10 days
   107→  customsSeaWorst: number;   // 18 days
   108→
   109→  // Customs (Air/Dhaka)
   110→  customsAirBest: number;    // 2 days
   111→  customsAirLikely: number;  // 3 days
   112→  customsAirWorst: number;   // 5 days
   113→}
   114→
   115→export const DEFAULT_LT_DISTRIBUTION: LeadTimeDistribution = {
   116→  mfgBest: 75, mfgLikely: 90, mfgWorst: 120,
   117→  shipSeaBest: 45, shipSeaLikely: 52, shipSeaWorst: 65,
   118→  shipAirBest: 5, shipAirLikely: 8, shipAirWorst: 12,
   119→  customsSeaBest: 7, customsSeaLikely: 10, customsSeaWorst: 18,
   120→  customsAirBest: 2, customsAirLikely: 3, customsAirWorst: 5,
   121→};
   122→
   123→/** PERT expected value: (best + 4*likely + worst) / 6 */
   124→function pertExpected(best: number, likely: number, worst: number): number {
   125→  return (best + 4 * likely + worst) / 6;
   126→}
   127→
   128→/** Approximate P90 using PERT: mean + 1.28 * stddev */
   129→function pertP90(best: number, likely: number, worst: number): number {
   130→  const mean = pertExpected(best, likely, worst);
   131→  const stddev = (worst - best) / 6;
   132→  return mean + 1.28 * stddev;
   133→}
   134→
   135→export function calculateLeadTimeP90(
   136→  dist: LeadTimeDistribution = DEFAULT_LT_DISTRIBUTION,
   137→  mode: 'sea' | 'air' = 'sea'
   138→): { totalP90: number; mfgP90: number; shipP90: number; customsP90: number } {
   139→  const mfgP90 = pertP90(dist.mfgBest, dist.mfgLikely, dist.mfgWorst);
   140→  const shipP90 = mode === 'sea'
   141→    ? pertP90(dist.shipSeaBest, dist.shipSeaLikely, dist.shipSeaWorst)
   142→    : pertP90(dist.shipAirBest, dist.shipAirLikely, dist.shipAirWorst);
   143→  const customsP90 = mode === 'sea'
   144→    ? pertP90(dist.customsSeaBest, dist.customsSeaLikely, dist.customsSeaWorst)
   145→    : pertP90(dist.customsAirBest, dist.customsAirLikely, dist.customsAirWorst);
   146→
   147→  return { totalP90: Math.round(mfgP90 + shipP90 + customsP90), mfgP90: Math.round(mfgP90), shipP90: Math.round(shipP90), customsP90: Math.round(customsP90) };
   148→}

   224→// Section 3.2: CNY Risk Assessment (Full)

   227→export type CNYStrategy = 'before_cny' | 'after_cny' | 'partial_order' | 'air_escape' | 'none';
   228→
   229→export interface CNYRiskAssessment {
   230→  hasRisk: boolean;
   231→  overlapDays: number;
   232→  effectiveCnyStart: Date;
   233→  cnyShutdownStart: Date;
   234→  cnyShutdownEnd: Date;
   235→  strategy: CNYStrategy;
   236→  additionalDelayDays: number;
   237→  latestSafeOrderDate: Date | null;
   238→  postCnyOrderDate: Date | null;
   239→  explanation: string;
   240→}
   241→
   242→function assessCNYRisk(
   243→  orderTriggerDate: Date,
   244→  mfgDays: number,
   245→  cnyInfo: CNYInfo | null,
   246→  today: Date,
   247→): CNYRiskAssessment {
   248→  if (!cnyInfo) {
   249→    return {
   250→      hasRisk: false, overlapDays: 0,
   251→      effectiveCnyStart: today, cnyShutdownStart: today, cnyShutdownEnd: today,
   252→      strategy: 'none', additionalDelayDays: 0,
   253→      latestSafeOrderDate: null, postCnyOrderDate: null,
   254→      explanation: 'No CNY window provided for analysis.',
   255→    };
   256→  }
   257→
   258→  const effectiveCnyStart = cnyInfo.effectiveStart;
   259→  const effectiveCnyEnd = cnyInfo.effectiveEnd;
   260→
   261→  // Manufacturing period
   262→  const orderProcDays = 2;
   263→  const mfgStart = addDays(orderTriggerDate, orderProcDays);
   264→  const mfgEnd = addDays(mfgStart, mfgDays);
   265→
   266→  // Check overlap
   267→  const overlapStart = new Date(Math.max(mfgStart.getTime(), effectiveCnyStart.getTime()));
   268→  const overlapEnd = new Date(Math.min(mfgEnd.getTime(), effectiveCnyEnd.getTime()));
   269→
   270→  if (overlapStart < overlapEnd) {
   271→    const overlapDays = Math.round((overlapEnd.getTime() - overlapStart.getTime()) / (1000 * 60 * 60 * 24));
   272→
   273→    // Strategy A: Latest safe order date
   274→    const latestSafe = addDays(effectiveCnyStart, -(mfgDays + orderProcDays));
   275→
   276→    // Strategy B: Post-CNY order date
   277→    const postCnyOrder = effectiveCnyEnd;
   278→
   279→    // Total CNY shutdown period
   280→    const cnyTotalDays = Math.round((effectiveCnyEnd.getTime() - effectiveCnyStart.getTime()) / (1000 * 60 * 60 * 24));
   281→
   282→    if (latestSafe >= today) {
   283→      return {
   284→        hasRisk: true, overlapDays,
   285→        effectiveCnyStart, cnyShutdownStart: cnyInfo.startDate, cnyShutdownEnd: cnyInfo.endDate,
   286→        strategy: 'before_cny', additionalDelayDays: 0,
   287→        latestSafeOrderDate: latestSafe, postCnyOrderDate: null,
   288→        explanation: `CNY RISK: Mfg overlaps CNY by ${overlapDays} days. STRATEGY A: Order by ${formatDate(latestSafe)} to complete manufacturing before CNY shutdown (${formatDate(cnyInfo.startDate)}–${formatDate(cnyInfo.endDate)}).`,
   289→      };
   290→    } else {
   291→      return {
   292→        hasRisk: true, overlapDays,
   293→        effectiveCnyStart, cnyShutdownStart: cnyInfo.startDate, cnyShutdownEnd: cnyInfo.endDate,
   294→        strategy: 'after_cny', additionalDelayDays: cnyTotalDays,
   295→        latestSafeOrderDate: null, postCnyOrderDate: postCnyOrder,
   296→        explanation: `CNY RISK: Mfg overlaps CNY by ${overlapDays} days. Cannot finish before CNY. STRATEGY B: Order after CNY on ${formatDate(postCnyOrder)}. Effective delay: ${cnyTotalDays} days.`,
   297→      };
   298→    }
   299→  }
   300→
   301→  return {
   302→    hasRisk: false, overlapDays: 0,
   303→    effectiveCnyStart, cnyShutdownStart: cnyInfo.startDate, cnyShutdownEnd: cnyInfo.endDate,
   304→    strategy: 'none', additionalDelayDays: 0,
   305→    latestSafeOrderDate: null, postCnyOrderDate: null,
   306→    explanation: `No CNY risk. Manufacturing completes before CNY window (${formatDate(cnyInfo.startDate)}–${formatDate(cnyInfo.endDate)}).`,
   307→  };
   308→}

   310→// =============================================
   311→// Section 3.3: CNY Strategy Selection Algorithm
   312→// =============================================
   313→
   314→export function selectCNYStrategy(
   315→  daysUntilStockout: number,
   316→  cnyDelayDays: number,
   317→  itemMarginPct: number,
   318→  itemUrgency: 'critical' | 'high' | 'normal' | 'low',
   319→  canAirShip: boolean,
   320→  airCostMultiplier: number = 8.0,
   321→): CNYStrategy {
   322→  // Can we survive the CNY delay with current stock?
   323→  if (daysUntilStockout > cnyDelayDays + 30) {
   324→    return 'after_cny';
   325→  }
   326→
   327→  // Stockout risk — try to order before CNY
   328→  if (itemUrgency === 'critical' || itemUrgency === 'high') {
   329→    return 'before_cny';
   330→  }
   331→
   332→  // Emergency: stockout imminent, consider air
   333→  if (daysUntilStockout <= 30 && canAirShip && itemMarginPct >= 30) {
   334→    return 'air_escape';
   335→  }
   336→
   337→  // Default: partial approach
   338→  return 'partial_order';
   339→}

   365→function buildTimeline(
   366→  orderTriggerDate: Date,
   367→  mfgDays: number,
   368→  shipmentDays: number,
   369→  customsDays: number,
   370→  orderProcessingDays: number,
   371→  packingLoadingDays: number,
   372→  warehouseTransportDays: number,
   373→  qcShelvingDays: number,
   374→  cnyDelayDays: number,
   375→): OrderTimeline {
   376→  const orderProcessingEnd = addDays(orderTriggerDate, orderProcessingDays);
   377→
   378→  const effectiveMfgDays = mfgDays + cnyDelayDays;
   379→  const mfgStartDate = orderProcessingEnd;
   380→  const mfgCompleteDate = addDays(mfgStartDate, effectiveMfgDays);
   381→
   382→  const packingLoadingEnd = addDays(mfgCompleteDate, packingLoadingDays);
   383→
   384→  const shipDepartureDate = packingLoadingEnd;
   385→  const arrivalDate = addDays(shipDepartureDate, shipmentDays);
   386→
   387→  const customsStartDate = arrivalDate;
   388→  const customsClearanceDate = addDays(customsStartDate, customsDays);
   389→
   390→  const warehouseArrivalDate = addDays(customsClearanceDate, warehouseTransportDays);
   391→  const availableForSaleDate = addDays(warehouseArrivalDate, qcShelvingDays);
   392→
   393→  const totalLeadTimeDays = orderProcessingDays + effectiveMfgDays + packingLoadingDays +
   394→    shipmentDays + customsDays + warehouseTransportDays + qcShelvingDays;
   395→
   396→  return {
   397→    orderTriggerDate,
   398→    orderProcessingEnd,
   399→    mfgStartDate,
   400→    mfgCompleteDate,
   401→    packingLoadingEnd,
   402→    shipDepartureDate,
   403→    arrivalDate,
   404→    customsStartDate,
   405→    customsClearanceDate,
   406→    warehouseArrivalDate,
   407→    availableForSaleDate,
   408→    totalLeadTimeDays,
   409→    cnyDelayDays,
   410→  };
   411→}

   417→export interface QuantityBreakdown {
   418→  totalNeeded: number;
   419→  totalSupply: number;
   420→  gap: number;
   421→  eoq: number;
   422→  moq: number;
   423→  maxStock: number;
   424→  preConstraintQty: number;
   425→  recommendedQty: number;
   426→  status: 'order_needed' | 'adequate' | 'overstock' | 'warehouse_full';
   427→  constraintsApplied: string[];
   428→  reason: string;
   429→}
   430→
   431→export function calculateRecommendedQty(
   432→  forecastedDemand: number,
   433→  safetyStock: number,
   434→  currentStock: number,
   435→  qtyOnOrder: number,
   436→  eoq: number,
   437→  moq: number,
   438→  maxStock: number,
   439→): QuantityBreakdown {
   440→  const constraintsApplied: string[] = [];
   441→
   442→  // Step 1-3: Calculate demand gap
   443→  const totalNeeded = forecastedDemand + safetyStock;
   444→  const totalSupply = currentStock + qtyOnOrder;
   445→  const gap = totalNeeded - totalSupply;
   446→
   447→  // Step 4: Check if stock is adequate
   448→  if (gap <= 0) {
   449→    return {
   450→      recommendedQty: 0, status: 'adequate',
   451→      reason: `Stock sufficient. Supply (${totalSupply}) covers need (${totalNeeded}).`,
   452→      gap, eoq, moq, maxStock, constraintsApplied: [],
   453→      totalNeeded, totalSupply, preConstraintQty: 0,
   454→    };
   455→  }
   456→
   457→  // Step 5: Check overstock
   458→  if (currentStock > maxStock) {
   459→    return {
   460→      recommendedQty: 0, status: 'overstock',
   461→      reason: `Currently overstocked (${currentStock} > max ${maxStock}). Allow draw-down.`,
   462→      gap, eoq, moq, maxStock, constraintsApplied: ['overstock_drawdown'],
   463→      totalNeeded, totalSupply, preConstraintQty: 0,
   464→    };
   465→  }
   466→
   467→  // Step 6: Start with economic quantity
   468→  let qty = Math.max(gap, eoq);
   469→  if (eoq > gap) {
   470→    constraintsApplied.push(`eoq_floor: EOQ (${eoq}) > gap (${gap}), using EOQ`);
   471→  }
   472→
   473→  // Step 7: Apply MOQ constraint
   474→  if (qty < moq) {
   475→    qty = moq;
   476→    constraintsApplied.push(`moq_floor: qty raised to MOQ (${moq})`);
   477→  }
   478→
   479→  const preConstraintQty = qty;
   480→
   481→  // Step 8: Apply warehouse capacity constraint
   482→  const warehouseCapacityRemaining = maxStock - currentStock;
   483→  if (qty > warehouseCapacityRemaining) {
   484→    qty = warehouseCapacityRemaining;
   485→    constraintsApplied.push(
   486→      `warehouse_cap: qty reduced from ${preConstraintQty} to ${qty} (max_stock ${maxStock} - current ${currentStock})`
   487→    );
   488→  }
   489→
   490→  // Step 9: Check if warehouse can't even fit MOQ
   491→  if (qty < moq && qty > 0) {
   492→    return {
   493→      recommendedQty: 0, status: 'warehouse_full',
   494→      reason: `Insufficient warehouse space for MOQ. Available space: ${warehouseCapacityRemaining}, MOQ: ${moq}.`,
   495→      gap, eoq, moq, maxStock, constraintsApplied: [...constraintsApplied, 'warehouse_below_moq'],
   496→      totalNeeded, totalSupply, preConstraintQty,
   497→    };
   498→  }
   499→
   500→  qty = Math.max(0, qty);
   501→  const status = qty > 0 ? 'order_needed' as const : 'adequate' as const;
   502→  const reason = `Order ${qty} units. Gap: ${gap}, constrained by: ${constraintsApplied.length > 0 ? constraintsApplied.join('; ') : 'none'}`;
   503→
   504→  return {
   505→    recommendedQty: qty, status, reason,
   506→    gap, eoq, moq, maxStock, constraintsApplied,
   507→    totalNeeded, totalSupply, preConstraintQty,
   508→  };
   509→}

   531→export function applySeasonalWeight(baseDemand: number, season: BDSeason, category: string): number {
   532→  const categoryAdj = CATEGORY_SEASONAL_ADJUSTMENTS[category];
   533→  if (categoryAdj && categoryAdj[season]) {
   534→    return baseDemand * categoryAdj[season];
   535→  }
   536→  return baseDemand * (SEASONAL_WEIGHTS[season] || 1.0);
   537→}
   538→
   539→export function getSeasonForDate(targetDate: Date): BDSeason {
   540→  const month = targetDate.getMonth() + 1;
   541→  if (month === 11 || month === 12 || month === 1 || month === 2) return 'winter';
   542→  if (month >= 3 && month <= 5) return 'summer';
   543→  if (month >= 6 && month <= 9) return 'monsoon';
   544→  return 'pre_winter';
   545→}
   546→
   547→export function getSeasonDateRange(season: BDSeason, year: number): { start: Date; end: Date } {
   548→  switch (season) {
   549→    case 'winter': return { start: new Date(year, 10, 1), end: new Date(year + 1, 1, 28) };
   550→    case 'summer': return { start: new Date(year, 2, 1), end: new Date(year, 4, 31) };
   551→    case 'monsoon': return { start: new Date(year, 5, 1), end: new Date(year, 8, 30) };
   552→    case 'pre_winter': return { start: new Date(year, 9, 1), end: new Date(year, 9, 31) };
   553→  }
   554→}

   628→/**
   629→ * THE CORE ALGORITHM — Full 9-step implementation
   630→ * Calculates when, what, and how much to order for a single SKU.
   631→ */
   632→export function calculateOrderTrigger(input: OrderTriggerInput): OrderTriggerResult {
   633→  const {
   634→    productId, productSku, productName, category,
   635→    currentStock, reservedStock, safetyStock, maxStock = 500,
   636→    reorderPoint, avgDailyDemand,
   637→    forecastedDemand = 0, qtyOnOrder = 0,
   638→    eoq = 100, moq = 50,
   639→    serviceLevel = 0.95,
   640→    bufferDays = 7,
   641→    monsoonAdjustment = 0,
   642→  } = input;
   643→
   644→  const availableStock = currentStock - reservedStock;
   645→  const today = new Date();
   646→  today.setHours(0, 0, 0, 0);
   647→
   648→  // Merge lead time config
   649→  const leadTimeConfig: LeadTimeConfig = { ...DEFAULT_LEAD_TIME, ...(input.leadTimeConfig || {}) };
   650→  if (input.shippingMethod) leadTimeConfig.shippingMethod = input.shippingMethod;
   651→
   652→  // ── STEP 1: Determine shipment and customs durations ──
   653→  const shipmentDays = leadTimeConfig.shippingMethod === 'sea'
   654→    ? leadTimeConfig.seaTransitDays + monsoonAdjustment
   655→    : leadTimeConfig.airTransitDays;
   656→  const customsDays = leadTimeConfig.shippingMethod === 'sea'
   657→    ? leadTimeConfig.customsClearanceDays
   658→    : 3;
   659→
   660→  // ── STEP 2: Calculate Reorder Point (Q1) ──
   661→  const totalLeadTime = leadTimeConfig.manufacturingDays + shipmentDays + customsDays;
   662→  const effectiveReorderPoint = reorderPoint > 0
   663→    ? reorderPoint
   664→    : (avgDailyDemand * totalLeadTime) + safetyStock;
   665→
   666→  // Determine stock status
   667→  let stockStatus: StockStatus;
   668→  let needsOrder: boolean;
   669→
   670→  if (availableStock <= 0) {
   671→    stockStatus = 'stockout';
   672→    needsOrder = true;
   673→  } else if (availableStock <= safetyStock) {
   674→    stockStatus = 'at_safety';
   675→    needsOrder = true;
   676→  } else if (availableStock <= effectiveReorderPoint) {
   677→    stockStatus = 'below_reorder';
   678→    needsOrder = true;
   679→  } else if (availableStock > maxStock) {
   680→    stockStatus = 'overstock';
   681→    needsOrder = false;
   682→  } else {
   683→    stockStatus = 'adequate';
   684→    needsOrder = availableStock < effectiveReorderPoint;
   685→  }
   686→
   687→  const daysOfStock = avgDailyDemand > 0
   688→    ? Math.floor(availableStock / avgDailyDemand)
   689→    : 999;
   690→
   691→  // ── STEP 3: Calculate Order Trigger Date (Q3) ──
   692→  let daysUntilSafetyStock: number;
   693→  if (avgDailyDemand > 0) {
   694→    daysUntilSafetyStock = (availableStock - safetyStock) / avgDailyDemand;
   695→  } else {
   696→    daysUntilSafetyStock = Infinity;
   697→  }
   698→
   699→  const safetyStockHitDate = addDays(today, Math.max(0, Math.round(daysUntilSafetyStock)));
   700→  let orderTriggerDate = addDays(safetyStockHitDate, -(totalLeadTime + bufferDays));
   701→
   702→  // ── STEP 4: CNY Risk Assessment ──
   703→  const cnyInfo = getCNYForDate(today);
   704→  let cnyRisk = assessCNYRisk(orderTriggerDate, leadTimeConfig.manufacturingDays, cnyInfo, today);
   705→
   706→  // If CNY risk detected, revise order trigger date
   707→  let cnyDelayDays = 0;
   708→  if (cnyRisk.hasRisk) {
   709→    if (cnyRisk.strategy === 'before_cny' && cnyRisk.latestSafeOrderDate && cnyRisk.latestSafeOrderDate >= today) {
   710→      // Order early enough to finish before CNY
   711→      orderTriggerDate = new Date(Math.min(orderTriggerDate.getTime(), cnyRisk.latestSafeOrderDate.getTime()));
   712→    } else if (cnyRisk.strategy === 'after_cny' && cnyRisk.postCnyOrderDate) {
   713→      // Accept delay, order after CNY
   714→      orderTriggerDate = cnyRisk.postCnyOrderDate;
   715→      cnyDelayDays = cnyRisk.additionalDelayDays;
   716→    }
   717→  }
   718→
   719→  // ── STEP 5: Build Complete Timeline ──
   720→  const timeline = buildTimeline(
   721→    orderTriggerDate,
   722→    leadTimeConfig.manufacturingDays,
   723→    shipmentDays,
   724→    customsDays,
   725→    leadTimeConfig.orderProcessingDays,
   726→    leadTimeConfig.packingLoadingDays,
   727→    leadTimeConfig.warehouseTransportDays,
   728→    leadTimeConfig.qcShelvingDays,
   729→    cnyDelayDays,
   730→  );
   731→
   732→  // ── STEP 6: Determine Urgency ──
   733→  const daysUntilTrigger = Math.round((orderTriggerDate.getTime() - today.getTime()) / (1000 * 60 * 60 * 24));
   734→
   735→  let urgency: Urgency;
   736→  if (daysUntilTrigger <= 0) urgency = 'critical';
   737→  else if (daysUntilTrigger <= 30) urgency = 'critical';
   738→  else if (daysUntilTrigger <= 90) urgency = 'high';
   739→  else if (daysUntilTrigger <= 180) urgency = 'normal';
   740→  else urgency = 'low';
   741→
   742→  // Upgrade urgency if CNY causes stockout gap
   743→  if (cnyRisk.hasRisk && urgency === 'normal' && cnyRisk.strategy === 'after_cny') {
   744→    const availableDate = timeline.availableForSaleDate;
   745→    if (safetyStockHitDate < availableDate) {
   746→      const stockoutDays = Math.round((availableDate.getTime() - safetyStockHitDate.getTime()) / (1000 * 60 * 60 * 24));
   747→      if (stockoutDays > 30) urgency = 'high';
   748→    }
   749→  }
   750→
   751→  // ── STEP 7: Calculate Recommended Order Quantity (Q2) ──
   752→  const qtyBreakdown = calculateRecommendedQty(
   753→    forecastedDemand,
   754→    safetyStock,
   755→    availableStock,
   756→    qtyOnOrder,
   757→    eoq,
   758→    moq,
   759→    maxStock,
   760→  );
   761→
   762→  // ── STEP 8: Recommend shipment mode ──
   763→  let recommendedShipmentMode: ShipmentMode = leadTimeConfig.shippingMethod;
   764→  if (urgency === 'critical' && leadTimeConfig.shippingMethod === 'sea' && daysUntilTrigger <= 0) {
   765→    recommendedShipmentMode = 'air';
   766→  }
   767→
   768→  // ── STEP 9: Return complete result ──
   769→  const currentSeason = getBDSeason(today.getMonth() + 1);
   770→  const seasonNote = getSeasonNote(currentSeason.season);
   771→
   772→  const leadTime = calculateTotalLeadTime(leadTimeConfig);
   773→
   774→  return {
   775→    productId, productSku, productName,
   776→    needsOrder, reorderPoint: Math.round(effectiveReorderPoint),
   777→    currentStock, availableStock, safetyStock, stockStatus, daysOfStock,
   778→    suggestedOrderQty: qtyBreakdown.recommendedQty,
   779→    qtyBreakdown,
   780→    orderTriggerDate, reorderHitDate: safetyStockHitDate,
   781→    expectedDeliveryDate: timeline.availableForSaleDate,
   782→    timeline, urgency, daysUntilTrigger,
   783→    totalLeadTimeDays: timeline.totalLeadTimeDays,
   784→    leadTimeBreakdown: leadTime.breakdown,
   785→    cnyRisk,
   786→    recommendedShipmentMode,
   787→    currentSeason: currentSeason.season,
   788→    seasonNote,
   789→  };
   790→}

   902→/**
   903→ * Parallel batch calculation with chunked concurrency control.
   904→ * For large catalogs (500+ SKUs), this provides significant speedup
   905→ * by processing chunks in parallel while maintaining memory safety.
   906→ *
   907→ * @param inputs - Array of OrderTriggerInput
   908→ * @param chunkSize - Number of products per parallel chunk (default: 20)
   909→ * @returns Sorted results by urgency
   910→ */
   911→export async function calculateBatchOrderTriggersParallel(
   912→  inputs: OrderTriggerInput[],
   913→  chunkSize: number = 20,
   914→): Promise<OrderTriggerResult[]> {
   915→  if (inputs.length <= chunkSize) {
   916→    // Small catalog — process synchronously for zero overhead
   917→    return calculateBatchOrderTriggers(inputs);
   918→  }
   919→
   920→  const chunks: OrderTriggerInput[][] = [];
   921→  for (let i = 0; i < inputs.length; i += chunkSize) {
   922→    chunks.push(inputs.slice(i, i + chunkSize));
   923→  }
   924→
   925→  // Process each chunk as a microtask (non-blocking)
   926→  const chunkResults = await Promise.all(
   927→    chunks.map(chunk =>
   928→      new Promise<OrderTriggerResult[]>(resolve => {
   929→        // Use setImmediate-like pattern via Promise.resolve
   930→        resolve(chunk.map(i => safeCalculateOrderTrigger(i)));
   931→      })
   932→    )
   933→  );
   934→
   935→  // Flatten and sort
   936→  const results = chunkResults.flat();
   937→  results.sort((a, b) => {
   938→    const priorityOrder = { critical: 0, high: 1, normal: 2, low: 3 };
   939→    return priorityOrder[a.urgency] - priorityOrder[b.urgency];
   940→  });
   941→
   942→  return results;
   943→}

   993→export function runOrderTriggerPipeline(
   994→  tenantId: string,
   995→  products: OrderTriggerInput[],
   996→  targetSeason: BDSeason,
   997→  targetYear: number,
   998→  unitCosts: Record<string, number> = {},
   999→  topN: number = 50,
  1000→): PipelineResult {
  1001→  const today = new Date();
  1002→  today.setHours(0, 0, 0, 0);
  1003→
  1004→  const seasonRange = getSeasonDateRange(targetSeason, targetYear);
  1005→  const cnyInfo = getCNYForDate(today);
  1006→
  1007→  // Calculate triggers for all products
  1008→  const triggers = calculateBatchOrderTriggers(products);
  1009→
  1010→  // Build pipeline products with seasonal adjustments
  1011→  const pipelineProducts: PipelineProduct[] = triggers
  1012→    .filter(t => t.needsOrder)
  1013→    .map(t => {
  1014→      const seasonalWeight = t.currentSeason
  1015→        ? applySeasonalWeight(1.0, t.currentSeason, t.currentSeason)
  1016→        : 1.0;
  1017→      const baseDemand = t.suggestedOrderQty;
  1018→      const adjustedDemand = Math.round(baseDemand * seasonalWeight);
  1019→      const unitCost = unitCosts[t.productSku] || 0;
  1020→
  1021→      return {
  1022→        productSku: t.productSku,
  1023→        productName: t.productName,
  1024→        category: products.find(p => p.productSku === t.productSku)?.category,
  1025→        forecastedDemand: baseDemand,
  1026→        adjustedDemand,
  1027→        recommendedQty: t.suggestedOrderQty,
  1028→        orderTriggerDate: formatDate(t.orderTriggerDate),
  1029→        expectedAvailableDate: formatDate(t.expectedDeliveryDate),
  1030→        urgency: t.urgency,
  1031→        daysUntilTrigger: t.daysUntilTrigger,
  1032→        cnyRisk: t.cnyRisk.hasRisk,
  1033→        cnyStrategy: t.cnyRisk.strategy,
  1034→        totalLeadTimeDays: t.totalLeadTimeDays,
  1035→        unitCostBdt: unitCost,
  1036→        totalCostBdt: t.suggestedOrderQty * unitCost,
  1037→      };
  1038→    });
  1039→
  1040→  // Sort by urgency, then by adjusted demand
  1041→  const urgencyOrder: Record<string, number> = { critical: 0, high: 1, normal: 2, low: 3 };
  1042→  pipelineProducts.sort((a, b) => {
  1043→    const uDiff = (urgencyOrder[a.urgency] || 99) - (urgencyOrder[b.urgency] || 99);
  1044→    if (uDiff !== 0) return uDiff;
  1045→    return b.adjustedDemand - a.adjustedDemand;
  1046→  });
  1047→
  1048→  const topProducts = pipelineProducts.slice(0, topN);
  1049→
  1050→  // Summary
  1051→  const urgencyCounts = { critical: 0, high: 0, normal: 0, low: 0 };
  1052→  topProducts.forEach(p => { urgencyCounts[p.urgency]++; });
  1053→
  1054→  const orderDates = topProducts.filter(p => p.recommendedQty > 0).map(p => p.orderTriggerDate);
  1055→  const totalSpend = topProducts.reduce((sum, p) => sum + p.totalCostBdt, 0);
  1056→
  1057→  return {
  1058→    forecastSessionId: `fs_${formatDateCompact(today)}_${targetSeason}${targetYear}`,
  1059→    tenantId,
  1060→    season: `${targetSeason}_${targetYear}`,
  1061→    period: {
  1062→      start: formatDate(seasonRange.start),
  1063→      end: formatDate(seasonRange.end),
  1064→      totalDays: Math.round((seasonRange.end.getTime() - seasonRange.start.getTime()) / (1000 * 60 * 60 * 24)),
  1065→    },
  1066→    generatedAt: today.toISOString(),
  1067→    cnyWindow: cnyInfo ? {
  1068→      year: cnyInfo.year,
  1069→      shutdownStart: formatDate(cnyInfo.startDate),
  1070→      shutdownEnd: formatDate(cnyInfo.endDate),
  1071→    } : null,
  1072→    products: topProducts,
  1073→    summary: {
  1074→      totalProducts: topProducts.length,
  1075→      totalRecommendedUnits: topProducts.reduce((sum, p) => sum + p.recommendedQty, 0),
  1076→      totalRecommendedSpendBdt: totalSpend,
  1077→      criticalUrgencyCount: urgencyCounts.critical,
  1078→      highUrgencyCount: urgencyCounts.high,
  1079→      normalUrgencyCount: urgencyCounts.normal,
  1080→      lowUrgencyCount: urgencyCounts.low,
  1081→      cnyRiskCount: topProducts.filter(p => p.cnyRisk).length,
  1082→      earliestOrderDate: orderDates.length > 0 ? orderDates.sort()[0] : null,
  1083→      latestOrderDate: orderDates.length > 0 ? orderDates.sort().reverse()[0] : null,
  1084→    },
  1085→  };
  1086→}

  1092→export interface StockProjectionPoint {
  1093→  date: string;
  1094→  stock: number;
  1095→  event?: 'stockout' | 'safety_hit' | 'reorder_hit' | 'arrival' | 'cny_start' | 'cny_end';
  1096→  note?: string;
  1097→}
  1098→
  1099→export function calculateStockProjection(
  1100→  currentStock: number,
  1101→  avgDailyDemand: number,
  1102→  safetyStock: number,
  1103→  reorderPoint: number,
  1104→  orderArrivalDate: Date,
  1105→  orderQty: number,
  1106→  horizonDays: number = 180,
  1107→): StockProjectionPoint[] {
  1108→  const points: StockProjectionPoint[] = [];
  1109→  const today = new Date();
  1110→  today.setHours(0, 0, 0, 0);
  1111→
  1112→  let stock = currentStock;
  1113→  let hasArrived = false;
  1114→
  1115→  for (let d = 0; d <= horizonDays; d++) {
  1116→    const date = addDays(today, d);
  1117→
  1118→    // Check if order arrives today
  1119→    if (!hasArrived && date >= orderArrivalDate) {
  1120→      stock += orderQty;
  1121→      hasArrived = true;
  1122→      points.push({ date: formatDate(date), stock: Math.round(stock), event: 'arrival', note: `Order arrives: +${orderQty} units` });
  1123→      continue;
  1124→    }
  1125→
  1126→    // Daily consumption
  1127→    stock -= avgDailyDemand;
  1128→
  1129→    let event: StockProjectionPoint['event'];
  1130→    let note: string | undefined;
  1131→
  1132→    if (stock <= 0 && stock + avgDailyDemand > 0) {
  1133→      event = 'stockout'; note = 'Stockout!';
  1134→    } else if (stock <= safetyStock && stock + avgDailyDemand > safetyStock) {
  1135→      event = 'safety_hit'; note = 'Stock hits safety level';
  1136→    } else if (stock <= reorderPoint && stock + avgDailyDemand > reorderPoint) {
  1137→      event = 'reorder_hit'; note = 'Stock hits reorder point';
  1138→    }
  1139→
  1140→    points.push({ date: formatDate(date), stock: Math.round(Math.max(0, stock)), event, note });
  1141→  }
  1142→
  1143→  return points;
  1144→}

  1177→export function calculateCNYDelay(
  1178→  orderDate: Date,

  1181→  for (const cny of CNY_DATES) {

  1183→      const overlapStart = new Date(Math.max(manufacturingEnd.getTime(), cny.effectiveStart.getTime()));
  1184→      const overlapEnd = new Date(Math.min(orderDate.getTime() + (manufacturingEnd.getTime() - orderDate.getTime()), cny.effectiveEnd.getTime()));
  1185→
  1186→      const isBeforeRush = orderDate < cny.rushStart;
  1187→
  1188→      if (isBeforeRush) {
  1189→        return {
  1190→          delayDays: Math.round(cny.shutdownDays * 0.3),
  1191→          cnyInfo: cny,
  1192→          strategy: `RUSH ORDER: Place before ${formatDate(cny.rushStart)} to beat CNY shutdown.`,
  1193→        };
  1194→      } else if (orderDate >= cny.endDate) {
  1195→        return {
  1196→          delayDays: 0,
  1197→          cnyInfo: cny,
  1198→          strategy: `POST-CNY: Order after factories reopen (${formatDate(cny.endDate)}).`,
  1199→        };
  1200→      } else {
  1201→        const daysToReopen = Math.max(0, Math.round((cny.effectiveEnd.getTime() - orderDate.getTime()) / (1000 * 60 * 60 * 24)));
  1202→        return {
  1203→          delayDays: daysToReopen,
  1204→          cnyInfo: cny,
  1205→          strategy: `CNY DELAY: ~${daysToReopen} days due to CNY shutdown (${formatDate(cny.startDate)} to ${formatDate(cny.endDate)}).`,
  1206→        };
  1207→      }
  1208→    }
  1209→  }
  1210→
  1211→  return { delayDays: 0, cnyInfo: null, strategy: 'No CNY impact detected.' };
