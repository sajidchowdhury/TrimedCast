    73→function ValidateStep() {
    74→  const { currentImportId, setLoading, setError, setValidationResult, setStep, isLoading } = useETLStore();
    75→  const handleProceed = useCallback(() => { setStep(3); }, [setStep]);
    76→
    77→  useEffect(() => {
    78→    const runValidation = async () => {
    79→      if (!currentImportId) return;
    80→      setLoading(true);
    81→      setError(null);
    82→      try {
    83→        const res = await fetch(`/api/imports/${currentImportId}/validate`, { method: 'POST' });
    84→        if (!res.ok) { const err = await res.json().catch(() => ({})); throw new Error(err.error || 'Validation failed'); }
    85→        const json = await res.json();
    86→        if (!json.success) throw new Error(json.error || 'Validation failed');
    87→        setValidationResult(json.data as ValidationResultClient);
    88→      } catch (err) { setError(err instanceof Error ? err.message : 'Validation failed'); }
    89→      finally { setLoading(false); }
    90→    };
    91→    runValidation();
    92→  }, [currentImportId, setLoading, setError, setValidationResult]);
    93→
    94→  return (
    95→    <div>
    96→      {isLoading && (
    97→        <Card className="mb-4">
    98→          <CardContent className="py-8 text-center">
    99→            <Loader2 className="h-8 w-8 mx-auto text-blue-500 animate-spin mb-3" />
   100→            <p className="text-sm text-gray-600">Running 3-phase validation...</p>
   101→            <p className="text-xs text-gray-400 mt-1">Schema → Data → Business Rules</p>
   102→          </CardContent>
   103→        </Card>
   104→      )}
   105→      <ValidationResults onProceed={handleProceed} />
   106→    </div>
   107→  );
   108→}

   110→function HarmonizeStep() {
   111→  const { currentImportId, setLoading, setError, setHarmonizationResult, setStep, harmonizationResult, isLoading } = useETLStore();
   112→  const handleProceed = useCallback(() => { setStep(4); }, [setStep]);
   113→
   114→  useEffect(() => {
   115→    if (harmonizationResult) return;
   116→    const runHarmonization = async () => {
   117→      if (!currentImportId) return;
   118→      setLoading(true); setError(null);
   119→      try {
   120→        const res = await fetch(`/api/imports/${currentImportId}/harmonize`, { method: 'POST' });
   121→        if (!res.ok) { const err = await res.json().catch(() => ({})); throw new Error(err.error || 'Harmonization failed'); }
   122→        const json = await res.json();
   123→        if (!json.success) throw new Error(json.error || 'Harmonization failed');
   124→        setHarmonizationResult(json.data as HarmonizationResultClient);
   125→      } catch (err) { setError(err instanceof Error ? err.message : 'Harmonization failed'); }
   126→      finally { setLoading(false); }
   127→    };
   128→    runHarmonization();
   129→  }, [currentImportId, harmonizationResult, setLoading, setError, setHarmonizationResult]);
   130→
   131→  if (isLoading && !harmonizationResult) {
   132→    return (
   133→      <Card>
   134→        <CardContent className="py-8 text-center">
   135→          <Loader2 className="h-8 w-8 mx-auto text-blue-500 animate-spin mb-3" />
   136→          <p className="text-sm text-gray-600">Running 6-step harmonization...</p>
   137→          <p className="text-xs text-gray-400 mt-1">Trim → Dates → Categories → Units → Dedup → Enrich</p>
   138→        </CardContent>
   139→      </Card>
   140→    );
   141→  }
   142→  return <HarmonizationLog onProceed={handleProceed} />;
   143→}

   145→function InsertStep() {
   146→  const { currentImportId, setLoading, setError, setInsertionResult, setStep, insertionResult, uploadResult, mappings, importType, isLoading } = useETLStore();
   147→
   148→  useEffect(() => {
   149→    if (insertionResult) return;
   150→    const runInsert = async () => {
   151→      if (!currentImportId) return;
   152→      setLoading(true); setError(null);
   153→      try {
   154→        const res = await fetch(`/api/imports/${currentImportId}/insert`, { method: 'POST' });
   155→        if (!res.ok) { const err = await res.json().catch(() => ({})); throw new Error(err.error || 'Insertion failed'); }
   156→        const json = await res.json();
   157→        if (!json.success) throw new Error(json.error || 'Insertion failed');
   158→        setInsertionResult(json.data as InsertionResultClient);
   159→      } catch (err) { setError(err instanceof Error ? err.message : 'Insertion failed'); }
   160→      finally { setLoading(false); }
   161→    };
   162→    runInsert();
   163→  }, [currentImportId, insertionResult, setLoading, setError, setInsertionResult]);
   164→
   165→  if (isLoading && !insertionResult) {
   166→    return (
   167→      <Card>
   168→        <CardContent className="py-8 text-center">
   169→          <Loader2 className="h-8 w-8 mx-auto text-blue-500 animate-spin mb-3" />
   170→          <p className="text-sm text-gray-600">Inserting data into database...</p>
   171→        </CardContent>
   172→      </Card>
   173→    );
   174→  }
   175→  if (!insertionResult) return null;
   176→
   177→  const schema = importType ? IMPORT_TYPE_SCHEMAS[importType as ImportType] : null;
   178→  const requiredTotal = schema?.requiredFields.length || 0;
   179→  const requiredMapped = mappings.filter((m) => m.targetField && m.confidence > 0 && schema?.requiredFields.some((rf) => rf.field === m.targetField)).length;
   180→  const qualityStats: QualityStats = {
   181→    rowsTotal: uploadResult?.rowsTotal || 0,
   182→    rowsValid: uploadResult?.rowsTotal || insertionResult.inserted,
   183→    rowsInserted: insertionResult.inserted,
   184→    rowsDuplicate: 0,
   185→    requiredMapped,
   186→    requiredTotal,
   187→  };
   188→
   189→  return (
   190→    <motion.div initial={{ opacity: 0, y: 20 }} animate={{ opacity: 1, y: 0 }} transition={{ duration: 0.4 }} className="space-y-4">
   191→      <Card className="border-emerald-200 bg-emerald-50/50">
   192→        <CardContent className="py-6">
   193→          <div className="flex items-center gap-3 mb-4">
   194→            <div className="w-12 h-12 rounded-full bg-emerald-100 flex items-center justify-center">
   195→              <CheckCircle2 className="h-7 w-7 text-emerald-600" />
   196→            </div>
   197→            <div>
   198→              <h2 className="text-xl font-bold text-emerald-800">Import Complete!</h2>
   199→              <p className="text-sm text-emerald-600">Your data has been successfully processed and inserted</p>
   200→            </div>
   201→          </div>
   202→          <QualityBadge score={insertionResult.qualityScore} size="lg" showBreakdown={true} stats={qualityStats} />
   203→        </CardContent>
   204→      </Card>
   205→      <div className="grid grid-cols-2 sm:grid-cols-4 gap-3">
   206→        <Card><CardContent className="pt-4 pb-3 text-center"><Database className="h-5 w-5 mx-auto text-blue-500 mb-1" /><div className="text-2xl font-bold text-gray-700">{insertionResult.inserted}</div><div className="text-xs text-gray-500">Rows Inserted</div></CardContent></Card>
   207→        <Card><CardContent className="pt-4 pb-3 text-center"><XCircle className="h-5 w-5 mx-auto text-red-500 mb-1" /><div className="text-2xl font-bold text-gray-700">{insertionResult.skipped}</div><div className="text-xs text-gray-500">Rows Skipped</div></CardContent></Card>
   208→        <Card><CardContent className="pt-4 pb-3 text-center"><FileSpreadsheet className="h-5 w-5 mx-auto text-gray-500 mb-1" /><div className="text-2xl font-bold text-gray-700">{insertionResult.createdMasterData?.length || 0}</div><div className="text-xs text-gray-500">Master Data Created</div></CardContent></Card>
   209→        <Card><CardContent className="pt-4 pb-3 text-center"><Loader2 className="h-5 w-5 mx-auto text-gray-400 mb-1" style={{ animation: 'none' }} /><div className="text-2xl font-bold text-gray-700">{insertionResult.durationMs !== null ? insertionResult.durationMs < 1000 ? `${insertionResult.durationMs}ms` : `${(insertionResult.durationMs / 1000).toFixed(1)}s` : '—'}</div><div className="text-xs text-gray-500">Duration</div></CardContent></Card>
   210→      </div>
   211→      {insertionResult.createdMasterData && insertionResult.createdMasterData.length > 0 && (
   212→        <Card>
   213→          <CardHeader className="pb-2"><CardTitle className="text-sm font-medium text-gray-600">Auto-Created Master Data</CardTitle></CardHeader>
   214→          <CardContent><ul className="text-xs text-gray-500 space-y-1">{insertionResult.createdMasterData.map((item, i) => (<li key={i} className="flex items-center gap-2"><Badge variant="outline" className="text-xs h-5">New</Badge>{item}</li>))}</ul></CardContent>
   215→        </Card>
   216→      )}
   217→      {insertionResult.errors && insertionResult.errors.length > 0 && (
   218→        <Card>
   219→          <CardHeader className="pb-2"><CardTitle className="text-sm font-medium text-red-600">Insertion Errors ({insertionResult.errors.length})</CardTitle></CardHeader>
   220→          <CardContent><ul className="text-xs text-red-500 space-y-1">{insertionResult.errors.slice(0, 10).map((err, i) => (<li key={i}>Row {err.row}: {err.error}</li>))}{insertionResult.errors.length > 10 && <li className="text-gray-400">...and {insertionResult.errors.length - 10} more</li>}</ul></CardContent>
   221→        </Card>
   222→      )}
   223→    </motion.div>
   224→  );
   225→}
