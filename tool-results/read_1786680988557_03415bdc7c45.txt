    70→function ValidateStep() {
    71→  const { currentImportId, setLoading, setError, setValidationResult, setStep, isLoading } = useETLStore();
    72→  const handleProceed = useCallback(() => { setStep(3); }, [setStep]);
    73→
    74→  useEffect(() => {
    75→    const runValidation = async () => {
    76→      if (!currentImportId) return;
    77→      setLoading(true);
    78→      setError(null);
    79→      try {
    80→        const res = await fetch(`/api/imports/${currentImportId}/validate`, { method: 'POST' });
    81→        if (!res.ok) { const err = await res.json().catch(() => ({})); throw new Error(err.error || 'Validation failed'); }
    82→        const json = await res.json();
    83→        if (!json.success) throw new Error(json.error || 'Validation failed');
    84→        setValidationResult(json.data as ValidationResultClient);
    85→      } catch (err) { setError(err instanceof Error ? err.message : 'Validation failed'); }
    86→      finally { setLoading(false); }
    87→    };
    88→    runValidation();
    89→  }, [currentImportId, setLoading, setError, setValidationResult]);
    90→
    91→  return (
    92→    <div>
    93→      {isLoading && (
    94→        <Card className="mb-4">
    95→          <CardContent className="py-8 text-center">
    96→            <Loader2 className="h-8 w-8 mx-auto text-blue-500 animate-spin mb-3" />
    97→            <p className="text-sm text-gray-600">Running 3-phase validation...</p>
    98→            <p className="text-xs text-gray-400 mt-1">Schema → Data → Business Rules</p>
    99→          </CardContent>
   100→        </Card>
   101→      )}
   102→      <ValidationResults onProceed={handleProceed} />
   103→    </div>
   104→  );
   105→}

   107→function HarmonizeStep() {
   108→  const { currentImportId, setLoading, setError, setHarmonizationResult, setStep, harmonizationResult, isLoading } = useETLStore();
   109→  const handleProceed = useCallback(() => { setStep(4); }, [setStep]);
   110→
   111→  useEffect(() => {
   112→    if (harmonizationResult) return;
   113→    const runHarmonization = async () => {
   114→      if (!currentImportId) return;
   115→      setLoading(true); setError(null);
   116→      try {
   117→        const res = await fetch(`/api/imports/${currentImportId}/harmonize`, { method: 'POST' });
   118→        if (!res.ok) { const err = await res.json().catch(() => ({})); throw new Error(err.error || 'Harmonization failed'); }
   119→        const json = await res.json();
   120→        if (!json.success) throw new Error(json.error || 'Harmonization failed');
   121→        setHarmonizationResult(json.data as HarmonizationResultClient);
   122→      } catch (err) { setError(err instanceof Error ? err.message : 'Harmonization failed'); }
   123→      finally { setLoading(false); }
   124→    };
   125→    runHarmonization();
   126→  }, [currentImportId, harmonizationResult, setLoading, setError, setHarmonizationResult]);
   127→
   128→  if (isLoading && !harmonizationResult) {
   129→    return (
   130→      <Card>
   131→        <CardContent className="py-8 text-center">
   132→          <Loader2 className="h-8 w-8 mx-auto text-blue-500 animate-spin mb-3" />
   133→          <p className="text-sm text-gray-600">Running 6-step harmonization...</p>
   134→          <p className="text-xs text-gray-400 mt-1">Trim → Dates → Categories → Units → Dedup → Enrich</p>
   135→        </CardContent>
   136→      </Card>
   137→    );
   138→  }
   139→  return <HarmonizationLog onProceed={handleProceed} />;
   140→}

   142→function InsertStep() {
   143→  const { currentImportId, setLoading, setError, setInsertionResult, setStep, insertionResult, uploadResult, mappings, importType, isLoading } = useETLStore();
   144→
   145→  useEffect(() => {
   146→    if (insertionResult) return;
   147→    const runInsert = async () => {
   148→      if (!currentImportId) return;
   149→      setLoading(true); setError(null);
   150→      try {
   151→        const res = await fetch(`/api/imports/${currentImportId}/insert`, { method: 'POST' });
   152→        if (!res.ok) { const err = await res.json().catch(() => ({})); throw new Error(err.error || 'Insertion failed'); }
   153→        const json = await res.json();
   154→        if (!json.success) throw new Error(json.error || 'Insertion failed');
   155→        setInsertionResult(json.data as InsertionResultClient);
   156→      } catch (err) { setError(err instanceof Error ? err.message : 'Insertion failed'); }
   157→      finally { setLoading(false); }
   158→    };
   159→    runInsert();
   160→  }, [currentImportId, insertionResult, setLoading, setError, setInsertionResult]);
   161→
   162→  if (isLoading && !insertionResult) {
   163→    return (
   164→      <Card>
   165→        <CardContent className="py-8 text-center">
   166→          <Loader2 className="h-8 w-8 mx-auto text-blue-500 animate-spin mb-3" />
   167→          <p className="text-sm text-gray-600">Inserting data into database...</p>
   168→        </CardContent>
   169→      </Card>
   170→    );
   171→  }
   172→  if (!insertionResult) return null;
   173→
   174→  const schema = importType ? IMPORT_TYPE_SCHEMAS[importType as ImportType] : null;
   175→  const requiredTotal = schema?.requiredFields.length || 0;
   176→  const requiredMapped = mappings.filter((m) => m.targetField && m.confidence > 0 && schema?.requiredFields.some((rf) => rf.field === m.targetField)).length;
   177→  const qualityStats: QualityStats = {
   178→    rowsTotal: uploadResult?.rowsTotal || 0,
   179→    rowsValid: uploadResult?.rowsTotal || insertionResult.inserted,
   180→    rowsInserted: insertionResult.inserted,
   181→    rowsDuplicate: 0,
   182→    requiredMapped,
   183→    requiredTotal,
   184→  };
   185→
   186→  return (
   187→    <motion.div initial={{ opacity: 0, y: 20 }} animate={{ opacity: 1, y: 0 }} transition={{ duration: 0.4 }} className="space-y-4">
   188→      <Card className="border-emerald-200 bg-emerald-50/50">
   189→        <CardContent className="py-6">
   190→          <div className="flex items-center gap-3 mb-4">
   191→            <div className="w-12 h-12 rounded-full bg-emerald-100 flex items-center justify-center">
   192→              <CheckCircle2 className="h-7 w-7 text-emerald-600" />
   193→            </div>
   194→            <div>
   195→              <h2 className="text-xl font-bold text-emerald-800">Import Complete!</h2>
   196→              <p className="text-sm text-emerald-600">Your data has been successfully processed and inserted</p>
   197→            </div>
   198→          </div>
   199→          <QualityBadge score={insertionResult.qualityScore} size="lg" showBreakdown={true} stats={qualityStats} />
   200→        </CardContent>
   201→      </Card>
   202→      <div className="grid grid-cols-2 sm:grid-cols-4 gap-3">
   203→        <Card><CardContent className="pt-4 pb-3 text-center"><Database className="h-5 w-5 mx-auto text-blue-500 mb-1" /><div className="text-2xl font-bold text-gray-700">{insertionResult.inserted}</div><div className="text-xs text-gray-500">Rows Inserted</div></CardContent></Card>
   204→        <Card><CardContent className="pt-4 pb-3 text-center"><XCircle className="h-5 w-5 mx-auto text-red-500 mb-1" /><div className="text-2xl font-bold text-gray-700">{insertionResult.skipped}</div><div className="text-xs text-gray-500">Rows Skipped</div></CardContent></Card>
   205→        <Card><CardContent className="pt-4 pb-3 text-center"><FileSpreadsheet className="h-5 w-5 mx-auto text-gray-500 mb-1" /><div className="text-2xl font-bold text-gray-700">{insertionResult.createdMasterData?.length || 0}</div><div className="text-xs text-gray-500">Master Data Created</div></CardContent></Card>
   206→        <Card><CardContent className="pt-4 pb-3 text-center"><Loader2 className="h-5 w-5 mx-auto text-gray-400 mb-1" style={{ animation: 'none' }} /><div className="text-2xl font-bold text-gray-700">{insertionResult.durationMs !== null ? insertionResult.durationMs < 1000 ? `${insertionResult.durationMs}ms` : `${(insertionResult.durationMs / 1000).toFixed(1)}s` : '—'}</div><div className="text-xs text-gray-500">Duration</div></CardContent></Card>
   207→      </div>
   208→      {insertionResult.createdMasterData && insertionResult.createdMasterData.length > 0 && (
   209→        <Card>
   210→          <CardHeader className="pb-2"><CardTitle className="text-sm font-medium text-gray-600">Auto-Created Master Data</CardTitle></CardHeader>
   211→          <CardContent><ul className="text-xs text-gray-500 space-y-1">{insertionResult.createdMasterData.map((item, i) => (<li key={i} className="flex items-center gap-2"><Badge variant="outline" className="text-xs h-5">New</Badge>{item}</li>))}</ul></CardContent>
   212→        </Card>
   213→      )}
   214→      {insertionResult.errors && insertionResult.errors.length > 0 && (
   215→        <Card>
   216→          <CardHeader className="pb-2"><CardTitle className="text-sm font-medium text-red-600">Insertion Errors ({insertionResult.errors.length})</CardTitle></CardHeader>
   217→          <CardContent><ul className="text-xs text-red-500 space-y-1">{insertionResult.errors.slice(0, 10).map((err, i) => (<li key={i}>Row {err.row}: {err.error}</li>))}{insertionResult.errors.length > 10 && <li className="text-gray-400">...and {insertionResult.errors.length - 10} more</li>}</ul></CardContent>
   218→        </Card>
   219→      )}
   220→    </motion.div>
   221→  );
   222→}
