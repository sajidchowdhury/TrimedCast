    83→function ValidateStep() {
    84→  const { currentImportId, setLoading, setError, setValidationResult, setStep, isLoading } = useETLStore();
    85→  const handleProceed = useCallback(() => { setStep(3); }, [setStep]);
    86→
    87→  useEffect(() => {
    88→    const runValidation = async () => {
    89→      if (!currentImportId) return;
    90→      setLoading(true);
    91→      setError(null);
    92→      try {
    93→        const res = await fetch(`/api/imports/${currentImportId}/validate`, { method: 'POST' });
    94→        if (!res.ok) { const err = await res.json().catch(() => ({})); throw new Error(err.error || 'Validation failed'); }
    95→        const json = await res.json();
    96→        if (!json.success) throw new Error(json.error || 'Validation failed');
    97→        setValidationResult(json.data as ValidationResultClient);
    98→      } catch (err) { setError(err instanceof Error ? err.message : 'Validation failed'); }
    99→      finally { setLoading(false); }
   100→    };
   101→    runValidation();
   102→  }, [currentImportId, setLoading, setError, setValidationResult]);
   103→
   104→  return (
   105→    <div>
   106→      {isLoading && (
   107→        <Card className="mb-4">
   108→          <CardContent className="py-8 text-center">
   109→            <Loader2 className="h-8 w-8 mx-auto text-blue-500 animate-spin mb-3" />
   110→            <p className="text-sm text-gray-600">Running 3-phase validation...</p>
   111→            <p className="text-xs text-gray-400 mt-1">Schema → Data → Business Rules</p>
   112→          </CardContent>
   113→        </Card>
   114→      )}
   115→      <ValidationResults onProceed={handleProceed} />
   116→    </div>
   117→  );
   118→}

   120→function HarmonizeStep() {
   121→  const { currentImportId, setLoading, setError, setHarmonizationResult, setStep, harmonizationResult, isLoading } = useETLStore();
   122→  const handleProceed = useCallback(() => { setStep(4); }, [setStep]);
   123→
   124→  useEffect(() => {
   125→    if (harmonizationResult) return;
   126→    const runHarmonization = async () => {
   127→      if (!currentImportId) return;
   128→      setLoading(true); setError(null);
   129→      try {
   130→        const res = await fetch(`/api/imports/${currentImportId}/harmonize`, { method: 'POST' });
   131→        if (!res.ok) { const err = await res.json().catch(() => ({})); throw new Error(err.error || 'Harmonization failed'); }
   132→        const json = await res.json();
   133→        if (!json.success) throw new Error(json.error || 'Harmonization failed');
   134→        setHarmonizationResult(json.data as HarmonizationResultClient);
   135→      } catch (err) { setError(err instanceof Error ? err.message : 'Harmonization failed'); }
   136→      finally { setLoading(false); }
   137→    };
   138→    runHarmonization();
   139→  }, [currentImportId, harmonizationResult, setLoading, setError, setHarmonizationResult]);
   140→
   141→  if (isLoading && !harmonizationResult) {
   142→    return (
   143→      <Card>
   144→        <CardContent className="py-8 text-center">
   145→          <Loader2 className="h-8 w-8 mx-auto text-blue-500 animate-spin mb-3" />
   146→          <p className="text-sm text-gray-600">Running 6-step harmonization...</p>
   147→          <p className="text-xs text-gray-400 mt-1">Trim → Dates → Categories → Units → Dedup → Enrich</p>
   148→        </CardContent>
   149→      </Card>
   150→    );
   151→  }
   152→  return <HarmonizationLog onProceed={handleProceed} />;
   153→}

   155→function InsertStep() {
   156→  const { currentImportId, setLoading, setError, setInsertionResult, setStep, insertionResult, uploadResult, mappings, importType, isLoading } = useETLStore();
   157→
   158→  useEffect(() => {
   159→    if (insertionResult) return;
   160→    const runInsert = async () => {
   161→      if (!currentImportId) return;
   162→      setLoading(true); setError(null);
   163→      try {
   164→        const res = await fetch(`/api/imports/${currentImportId}/insert`, { method: 'POST' });
   165→        if (!res.ok) { const err = await res.json().catch(() => ({})); throw new Error(err.error || 'Insertion failed'); }
   166→        const json = await res.json();
   167→        if (!json.success) throw new Error(json.error || 'Insertion failed');
   168→        setInsertionResult(json.data as InsertionResultClient);
   169→      } catch (err) { setError(err instanceof Error ? err.message : 'Insertion failed'); }
   170→      finally { setLoading(false); }
   171→    };
   172→    runInsert();
   173→  }, [currentImportId, insertionResult, setLoading, setError, setInsertionResult]);
   174→
   175→  if (isLoading && !insertionResult) {
   176→    return (
   177→      <Card>
   178→        <CardContent className="py-8 text-center">
   179→          <Loader2 className="h-8 w-8 mx-auto text-blue-500 animate-spin mb-3" />
   180→          <p className="text-sm text-gray-600">Inserting data into database...</p>
   181→        </CardContent>
   182→      </Card>
   183→    );
   184→  }
   185→  if (!insertionResult) return null;
   186→
   187→  const schema = importType ? IMPORT_TYPE_SCHEMAS[importType as ImportType] : null;
   188→  const requiredTotal = schema?.requiredFields.length || 0;
   189→  const requiredMapped = mappings.filter((m) => m.targetField && m.confidence > 0 && schema?.requiredFields.some((rf) => rf.field === m.targetField)).length;
   190→  const qualityStats: QualityStats = {
   191→    rowsTotal: uploadResult?.rowsTotal || 0,
   192→    rowsValid: uploadResult?.rowsTotal || insertionResult.inserted,
   193→    rowsInserted: insertionResult.inserted,
   194→    rowsDuplicate: 0,
   195→    requiredMapped,
   196→    requiredTotal,
   197→  };
   198→
   199→  return (
   200→    <motion.div initial={{ opacity: 0, y: 20 }} animate={{ opacity: 1, y: 0 }} transition={{ duration: 0.4 }} className="space-y-4">
   201→      <Card className="border-emerald-200 bg-emerald-50/50">
   202→        <CardContent className="py-6">
   203→          <div className="flex items-center gap-3 mb-4">
   204→            <div className="w-12 h-12 rounded-full bg-emerald-100 flex items-center justify-center">
   205→              <CheckCircle2 className="h-7 w-7 text-emerald-600" />
   206→            </div>
   207→            <div>
   208→              <h2 className="text-xl font-bold text-emerald-800">Import Complete!</h2>
   209→              <p className="text-sm text-emerald-600">Your data has been successfully processed and inserted</p>
   210→            </div>
   211→          </div>
   212→          <QualityBadge score={insertionResult.qualityScore} size="lg" showBreakdown={true} stats={qualityStats} />
   213→        </CardContent>
   214→      </Card>
   215→      <div className="grid grid-cols-2 sm:grid-cols-4 gap-3">
   216→        <Card><CardContent className="pt-4 pb-3 text-center"><Database className="h-5 w-5 mx-auto text-blue-500 mb-1" /><div className="text-2xl font-bold text-gray-700">{insertionResult.inserted}</div><div className="text-xs text-gray-500">Rows Inserted</div></CardContent></Card>
   217→        <Card><CardContent className="pt-4 pb-3 text-center"><XCircle className="h-5 w-5 mx-auto text-red-500 mb-1" /><div className="text-2xl font-bold text-gray-700">{insertionResult.skipped}</div><div className="text-xs text-gray-500">Rows Skipped</div></CardContent></Card>
   218→        <Card><CardContent className="pt-4 pb-3 text-center"><FileSpreadsheet className="h-5 w-5 mx-auto text-gray-500 mb-1" /><div className="text-2xl font-bold text-gray-700">{insertionResult.createdMasterData?.length || 0}</div><div className="text-xs text-gray-500">Master Data Created</div></CardContent></Card>
   219→        <Card><CardContent className="pt-4 pb-3 text-center"><Loader2 className="h-5 w-5 mx-auto text-gray-400 mb-1" style={{ animation: 'none' }} /><div className="text-2xl font-bold text-gray-700">{insertionResult.durationMs !== null ? insertionResult.durationMs < 1000 ? `${insertionResult.durationMs}ms` : `${(insertionResult.durationMs / 1000).toFixed(1)}s` : '—'}</div><div className="text-xs text-gray-500">Duration</div></CardContent></Card>
   220→      </div>
   221→      {insertionResult.createdMasterData && insertionResult.createdMasterData.length > 0 && (
   222→        <Card>
   223→          <CardHeader className="pb-2"><CardTitle className="text-sm font-medium text-gray-600">Auto-Created Master Data</CardTitle></CardHeader>
   224→          <CardContent><ul className="text-xs text-gray-500 space-y-1">{insertionResult.createdMasterData.map((item, i) => (<li key={i} className="flex items-center gap-2"><Badge variant="outline" className="text-xs h-5">New</Badge>{item}</li>))}</ul></CardContent>
   225→        </Card>
   226→      )}
   227→      {insertionResult.errors && insertionResult.errors.length > 0 && (
   228→        <Card>
   229→          <CardHeader className="pb-2"><CardTitle className="text-sm font-medium text-red-600">Insertion Errors ({insertionResult.errors.length})</CardTitle></CardHeader>
   230→          <CardContent><ul className="text-xs text-red-500 space-y-1">{insertionResult.errors.slice(0, 10).map((err, i) => (<li key={i}>Row {err.row}: {err.error}</li>))}{insertionResult.errors.length > 10 && <li className="text-gray-400">...and {insertionResult.errors.length - 10} more</li>}</ul></CardContent>
   231→        </Card>
   232→      )}
   233→    </motion.div>
   234→  );
   235→}
