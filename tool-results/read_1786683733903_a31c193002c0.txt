    94→function formatDate(dateStr: string): string {
    95→  try {
    96→    const d = new Date(dateStr);
    97→    return d.toLocaleDateString('en-US', { month: 'short', day: 'numeric' });
    98→  } catch {
    99→    return dateStr;
   100→  }
   101→}
   102→
   103→function formatBDT(amount: number): string {
   104→  return '৳' + amount.toLocaleString('en-BD', { minimumFractionDigits: 0, maximumFractionDigits: 0 });
   105→}
   106→
   107→function stockColor(current: number, reorder: number): string {
   108→  if (current <= 0) return 'text-red-600';
   109→  if (current < reorder * 0.5) return 'text-red-500';
   110→  if (current < reorder) return 'text-amber-500';
   111→  return 'text-emerald-500';
   112→}

   205→export function RecommendedOrdersTable() {
   206→  // State
   207→  const [orders, setOrders] = useState<RecommendedOrderRow[]>([]);
   208→  const [summary, setSummary] = useState<SummaryStats | null>(null);
   209→  const [loading, setLoading] = useState(true);
   210→  const [convertingId, setConvertingId] = useState<string | null>(null);
   211→  const [expandedRow, setExpandedRow] = useState<string | null>(null);
   212→  const [page, setPage] = useState(1);
   213→
   214→  // Filters
   215→  const [priorityFilter, setPriorityFilter] = useState<string>('all');
   216→  const [seasonFilter, setSeasonFilter] = useState<string>('all');
   217→  const [cnyFilter, setCnyFilter] = useState<string>('all');
   218→  const [statusFilter, setStatusFilter] = useState<string>('all');
   219→  const [search, setSearch] = useState<string>('');
   220→
   221→  // Sort
   222→  const [sortField, setSortField] = useState<SortField>('priority');
   223→  const [sortDir, setSortDir] = useState<SortDir>('asc');
   224→
   225→  // Fetch orders
   226→  const fetchOrders = useCallback(async () => {
   227→    setLoading(true);
   228→    try {
   229→      const res = await fetch(`/api/orders/triggers?tenantId=${TENANT_ID}`, {
   230→        method: 'POST',
   231→        headers: { 'Content-Type': 'application/json' },
   232→        body: JSON.stringify({ tenantId: TENANT_ID, shippingMethod: 'sea', serviceLevel: 0.95 }),
   233→      });
   234→      if (res.ok) {
   235→        const json = await res.json();
   236→        if (json.success && json.data) {
   237→          const triggers = json.data.triggers || [];
   238→          const mapped: RecommendedOrderRow[] = triggers.map((t: Record<string, unknown>) => ({
   239→            id: (t.productId as string) + '-' + (t.productSku as string),
   240→            productId: t.productId as string,
   241→            productSku: (t.productSku as string) || '',
   242→            productName: (t.productName as string) || '',
   243→            category: '',
   244→            currentStock: (t.currentStock as number) || 0,
   245→            reorderPoint: (t.reorderPoint as number) || 0,
   246→            suggestedQty: (t.suggestedOrderQty as number) || 0,
   247→            orderTriggerDate: (t.orderTriggerDate as string) || '',
   248→            expectedDeliveryDate: (t.expectedDeliveryDate as string) || '',
   249→            totalLeadTime: (t.totalLeadTimeDays as number) || 0,
   250→            priority: (t.priority as string) || 'normal',
   251→            cnyRisk: (t.cnyRisk as boolean) || false,
   252→            season: (t.currentSeason as string) || '',
   253→            status: 'pending' as const,
   254→            orderTrigger: (t.orderTrigger as string) || '',
   255→          }));
   256→          setOrders(mapped);
   257→          if (json.data.summary) {
   258→            setSummary(json.data.summary as SummaryStats);
   259→          }
   260→        }
   261→      }
   262→    } catch {
   263→      // Silently fail - empty state will show
   264→    } finally {
   265→      setLoading(false);
   266→    }
   267→  }, []);
   268→
   269→  useEffect(() => { fetchOrders(); }, [fetchOrders]);
   270→
   271→  // Convert to PO
   272→  const handleConvert = useCallback(async (order: RecommendedOrderRow) => {
   273→    setConvertingId(order.id);
   274→    try {
   275→      const res = await fetch('/api/orders/convert', {
   276→        method: 'POST',
   277→        headers: { 'Content-Type': 'application/json' },
   278→        body: JSON.stringify({ tenantId: TENANT_ID, recommendedOrderId: order.id }),
   279→      });
   280→      if (res.ok) {
   281→        const json = await res.json();
   282→        if (json.success) {
   283→          setOrders(prev => prev.map(o => o.id === order.id ? { ...o, status: 'converted' as const } : o));
   284→        }
   285→      }
   286→    } catch {
   287→      // Silently fail
   288→    } finally {
   289→      setConvertingId(null);
   290→    }
   291→  }, []);
   292→
   293→  // Toggle sort
   294→  const handleSort = useCallback((field: SortField) => {
   295→    setSortField(prev => {
   296→      if (prev === field) {
   297→        setSortDir(d => d === 'asc' ? 'desc' : 'asc');
   298→        return prev;
   299→      }
   300→      setSortDir('asc');
   301→      return field;
   302→    });
   303→  }, []);
   304→
   305→  // Filtered + sorted data
   306→  const filtered = useMemo(() => {
   307→    let data = [...orders];
   308→    if (priorityFilter !== 'all') data = data.filter(o => o.priority === priorityFilter);
   309→    if (seasonFilter !== 'all') data = data.filter(o => o.season === seasonFilter);
   310→    if (cnyFilter === 'risk') data = data.filter(o => o.cnyRisk);
   311→    else if (cnyFilter === 'no_risk') data = data.filter(o => !o.cnyRisk);
   312→    if (statusFilter !== 'all') data = data.filter(o => o.status === statusFilter);
   313→    if (search.trim()) {
   314→      const q = search.toLowerCase();
   315→      data = data.filter(o => o.productSku.toLowerCase().includes(q) || o.productName.toLowerCase().includes(q));
   316→    }
   317→    data.sort((a, b) => {
   318→      let cmp = 0;
   319→      switch (sortField) {
   320→        case 'priority':
   321→          cmp = (PRIORITY_CONFIG[a.priority]?.order ?? 2) - (PRIORITY_CONFIG[b.priority]?.order ?? 2);
   322→          break;
   323→        case 'orderTriggerDate':
   324→          cmp = a.orderTriggerDate.localeCompare(b.orderTriggerDate);
   325→          break;
   326→        case 'suggestedQty':
   327→          cmp = a.suggestedQty - b.suggestedQty;
   328→          break;
   329→        case 'totalLeadTime':
   330→          cmp = a.totalLeadTime - b.totalLeadTime;
   331→          break;
   332→      }
   333→      return sortDir === 'asc' ? cmp : -cmp;
   334→    });
   335→    return data;
   336→  }, [orders, priorityFilter, seasonFilter, cnyFilter, statusFilter, search, sortField, sortDir]);
   337→
   338→  // Pagination
   339→  const totalPages = Math.max(1, Math.ceil(filtered.length / PAGE_SIZE));
   340→  const safePage = Math.min(page, totalPages);
   341→  const paged = filtered.slice((safePage - 1) * PAGE_SIZE, safePage * PAGE_SIZE);
   342→
   343→  const paginationRange = useMemo(() => {
   344→    const range: (number | 'ellipsis')[] = [];
   345→    if (totalPages <= 7) {
   346→      for (let i = 1; i <= totalPages; i++) range.push(i);
   347→    } else {
   348→      range.push(1);
   349→      if (safePage > 3) range.push('ellipsis');
   350→      const start = Math.max(2, safePage - 1);
   351→      const end = Math.min(totalPages - 1, safePage + 1);
   352→      for (let i = start; i <= end; i++) range.push(i);
   353→      if (safePage < totalPages - 2) range.push('ellipsis');
   354→      range.push(totalPages);
   355→    }
   356→    return range;
   357→  }, [totalPages, safePage]);
   358→
   359→  // Computed summary from filtered
   360→  const filteredSummary = useMemo(() => {
   361→    const totalUrgent = filtered.filter(o => o.priority === 'urgent').length;
   362→    const totalHigh = filtered.filter(o => o.priority === 'high').length;
   363→    const cnyRiskCount = filtered.filter(o => o.cnyRisk).length;
   364→    return { total: filtered.length, totalUrgent, totalHigh, cnyRiskCount };
   365→  }, [filtered]);
   366→
   367→  // Sort icon helper
   368→  function SortIcon({ field }: { field: SortField }) {
   369→    const isActive = sortField === field;
   370→    return (
   371→      <ArrowUpDown className={`h-3 w-3 inline ml-1 ${isActive ? 'text-emerald-600' : 'text-slate-400'}`} />
   372→    );
   373→  }
   374→
   375→  // ============================================
   376→  // Render
   377→  // ============================================
   378→
   379→  return (
   380→    <div className="space-y-4">
   381→      {/* ========== Summary Stats Row ========== */}
   382→      <div className="grid grid-cols-2 sm:grid-cols-5 gap-3">
   383→        <Card className="bg-white shadow-sm">
   384→          <CardContent className="p-3 flex items-center gap-3">
   385→            <div className="w-9 h-9 rounded-lg bg-slate-100 flex items-center justify-center">
   386→              <Package className="h-4 w-4 text-slate-600" />
   387→            </div>
   388→            <div>
   389→              <p className="text-[10px] uppercase tracking-wider text-slate-500">Total Orders</p>
   390→              <p className="text-lg font-bold text-slate-800">{filteredSummary.total}</p>
   391→            </div>
   392→          </CardContent>
   393→        </Card>
   394→
   395→        <Card className="bg-white shadow-sm">
   396→          <CardContent className="p-3 flex items-center gap-3">
   397→            <div className="w-9 h-9 rounded-lg bg-red-50 flex items-center justify-center">
   398→              <Flame className="h-4 w-4 text-red-500" />
   399→            </div>
   400→            <div>
   401→              <p className="text-[10px] uppercase tracking-wider text-slate-500">Urgent</p>
   402→              <p className="text-lg font-bold text-red-600">{filteredSummary.totalUrgent}</p>
   403→            </div>
   404→          </CardContent>
   405→        </Card>
   406→
   407→        <Card className="bg-white shadow-sm">
   408→          <CardContent className="p-3 flex items-center gap-3">
   409→            <div className="w-9 h-9 rounded-lg bg-orange-50 flex items-center justify-center">
   410→              <AlertTriangle className="h-4 w-4 text-orange-500" />
   411→            </div>
   412→            <div>
   413→              <p className="text-[10px] uppercase tracking-wider text-slate-500">High</p>
   414→              <p className="text-lg font-bold text-orange-600">{filteredSummary.totalHigh}</p>
   415→            </div>
   416→          </CardContent>
   417→        </Card>
   418→
   419→        <Card className="bg-white shadow-sm">
   420→          <CardContent className="p-3 flex items-center gap-3">
   421→            <div className="w-9 h-9 rounded-lg bg-emerald-50 flex items-center justify-center">
   422→              <TrendingUp className="h-4 w-4 text-emerald-500" />
   423→            </div>
   424→            <div>
   425→              <p className="text-[10px] uppercase tracking-wider text-slate-500">Suggested Spend</p>
   426→              <p className="text-lg font-bold text-emerald-700">
   427→                {summary ? formatBDT(summary.totalSuggestedSpend) : '—'}
   428→              </p>
   429→            </div>
   430→          </CardContent>
   431→        </Card>
   432→
   433→        <Card className="bg-white shadow-sm">
   434→          <CardContent className="p-3 flex items-center gap-3">
   435→            <div className="w-9 h-9 rounded-lg bg-red-50 flex items-center justify-center">
   436→              <ShieldAlert className="h-4 w-4 text-red-500" />
   437→            </div>
   438→            <div>
   439→              <p className="text-[10px] uppercase tracking-wider text-slate-500">CNY Risk</p>
   440→              <p className="text-lg font-bold text-red-600">{filteredSummary.cnyRiskCount}</p>
   441→            </div>
   442→          </CardContent>
   443→        </Card>
   444→      </div>
   445→
   446→      {/* ========== Filter Bar ========== */}
   447→      <div className="flex flex-wrap items-center gap-2">
   448→        <Select value={priorityFilter} onValueChange={setPriorityFilter}>
   449→          <SelectTrigger size="sm" className="w-[130px]">
   450→            <SelectValue placeholder="Priority" />
   451→          </SelectTrigger>
   452→          <SelectContent>
   453→            <SelectItem value="all">All Priorities</SelectItem>
   454→            <SelectItem value="urgent">Urgent</SelectItem>
   455→            <SelectItem value="high">High</SelectItem>
   456→            <SelectItem value="normal">Normal</SelectItem>
   457→            <SelectItem value="low">Low</SelectItem>
   458→          </SelectContent>
   459→        </Select>
   460→
   461→        <Select value={seasonFilter} onValueChange={setSeasonFilter}>
   462→          <SelectTrigger size="sm" className="w-[140px]">
   463→            <SelectValue placeholder="Season" />
   464→          </SelectTrigger>
   465→          <SelectContent>
   466→            <SelectItem value="all">All Seasons</SelectItem>
   467→            <SelectItem value="winter">Winter</SelectItem>
   468→            <SelectItem value="summer">Summer</SelectItem>
   469→            <SelectItem value="monsoon">Monsoon</SelectItem>
   470→            <SelectItem value="pre_winter">Pre-Winter</SelectItem>
   471→          </SelectContent>
   472→        </Select>
   473→
   474→        <Select value={cnyFilter} onValueChange={setCnyFilter}>
   475→          <SelectTrigger size="sm" className="w-[140px]">
   476→            <SelectValue placeholder="CNY Risk" />
   477→          </SelectTrigger>
   478→          <SelectContent>
   479→            <SelectItem value="all">All CNY Risk</SelectItem>
   480→            <SelectItem value="risk">CNY Risk Only</SelectItem>
   481→            <SelectItem value="no_risk">No CNY Risk</SelectItem>
   482→          </SelectContent>
   483→        </Select>
   484→
   485→        <Select value={statusFilter} onValueChange={setStatusFilter}>
   486→          <SelectTrigger size="sm" className="w-[130px]">
   487→            <SelectValue placeholder="Status" />
   488→          </SelectTrigger>
   489→          <SelectContent>
   490→            <SelectItem value="all">All Status</SelectItem>
   491→            <SelectItem value="pending">Pending</SelectItem>
   492→            <SelectItem value="approved">Approved</SelectItem>
   493→            <SelectItem value="converted">Converted</SelectItem>
   494→            <SelectItem value="rejected">Rejected</SelectItem>
   495→          </SelectContent>
   496→        </Select>
   497→
   498→        <div className="relative flex-1 min-w-[180px] max-w-[300px]">
   499→          <Search className="absolute left-2.5 top-1/2 -translate-y-1/2 h-3.5 w-3.5 text-slate-400" />
   500→          <Input
   501→            placeholder="Search SKU or product..."
   502→            value={search}
   503→            onChange={e => { setSearch(e.target.value); setPage(1); }}
   504→            className="h-8 pl-8 text-sm"
   505→          />
   506→        </div>
   507→
   508→        {filtered.length !== orders.length && (
   509→          <Badge variant="outline" className="text-xs text-slate-500">
   510→            {filtered.length} of {orders.length} orders
   511→          </Badge>
   512→        )}
   513→      </div>
   514→
   515→      {/* ========== Loading State ========== */}
   516→      {loading && (
   517→        <div className="flex items-center justify-center py-16 gap-2 text-slate-500">
   518→          <Loader2 className="h-5 w-5 animate-spin" />
   519→          <span className="text-sm">Calculating order triggers for all products...</span>
   520→        </div>
   521→      )}
   522→
   523→      {/* ========== Empty State ========== */}
   524→      {!loading && filtered.length === 0 && (
   525→        <div className="text-center py-16">
   526→          <Package className="h-10 w-10 text-slate-300 mx-auto mb-3" />
   527→          <p className="text-sm text-slate-500">
   528→            {orders.length === 0
   529→              ? 'No recommended orders found. Import sales data and generate forecasts first.'
   530→              : 'No orders match the current filters.'}
   531→          </p>
   532→        </div>
   533→      )}
   534→
   535→      {/* ========== Desktop Table View ========== */}
   536→      {!loading && filtered.length > 0 && (
   537→        <>
   538→          {/* Desktop: Table */}
   539→          <div className="hidden md:block rounded-lg border border-slate-200 bg-white overflow-hidden">
   540→            <Table>
   541→              <TableHeader>
   542→                <TableRow className="bg-slate-50/80 hover:bg-slate-50/80">
   543→                  <TableHead className="w-8" />
   544→                  <TableHead className="w-24">
   545→                    <button className="inline-flex items-center" onClick={() => handleSort('priority')}>
   546→                      Priority <SortIcon field="priority" />
   547→                    </button>
   548→                  </TableHead>
   549→                  <TableHead>Product</TableHead>
   550→                  <TableHead className="hidden lg:table-cell">Category</TableHead>
   551→                  <TableHead>Stock / ROP</TableHead>
   552→                  <TableHead>
   553→                    <button className="inline-flex items-center" onClick={() => handleSort('suggestedQty')}>
   554→                      Suggested Qty <SortIcon field="suggestedQty" />
   555→                    </button>
   556→                  </TableHead>
   557→                  <TableHead className="hidden xl:table-cell">
   558→                    <button className="inline-flex items-center" onClick={() => handleSort('orderTriggerDate')}>
   559→                      Order Date <SortIcon field="orderTriggerDate" />
   560→                    </button>
   561→                  </TableHead>
   562→                  <TableHead className="hidden xl:table-cell">Delivery</TableHead>
   563→                  <TableHead className="hidden lg:table-cell">
   564→                    <button className="inline-flex items-center" onClick={() => handleSort('totalLeadTime')}>
   565→                      Lead Time <SortIcon field="totalLeadTime" />
   566→                    </button>
   567→                  </TableHead>
   568→                  <TableHead className="hidden lg:table-cell">CNY</TableHead>
   569→                  <TableHead className="hidden xl:table-cell">Season</TableHead>
   570→                  <TableHead className="w-32">Actions</TableHead>
   571→                </TableRow>
   572→              </TableHeader>
   573→              <TableBody>
   574→                <AnimatePresence initial={false}>
   575→                  {paged.map(order => {
   576→                    const isExpanded = expandedRow === order.id;
   577→                    const pCfg = PRIORITY_CONFIG[order.priority] || PRIORITY_CONFIG.normal;
   578→                    const isConverting = convertingId === order.id;
   579→
   580→                    return (
   581→                      <motion.tr
   582→                        key={order.id}
   583→                        initial={{ opacity: 0, y: -4 }}
   584→                        animate={{ opacity: 1, y: 0 }}
   585→                        exit={{ opacity: 0, y: 4 }}
   586→                        transition={{ duration: 0.15 }}
   587→                        className={`group cursor-pointer ${pCfg.border} ${
   588→                          isExpanded ? 'bg-slate-50' : 'hover:bg-slate-50/50'
   589→                        }`}
   590→                        onClick={() => setExpandedRow(isExpanded ? null : order.id)}
   591→                      >
   592→                        {/* Expand toggle */}
   593→                        <TableCell className="w-8" onClick={e => e.stopPropagation()}>
   594→                          {isExpanded ? (
   595→                            <ChevronDown className="h-4 w-4 text-slate-400" />
   596→                          ) : (
   597→                            <ChevronRight className="h-4 w-4 text-slate-400" />
   598→                          )}
   599→                        </TableCell>
   600→
   601→                        {/* Priority */}
   602→                        <TableCell onClick={e => e.stopPropagation()}>
   603→                          <Badge className={pCfg.bg}>{pCfg.label}</Badge>
   604→                        </TableCell>
   605→
   606→                        {/* Product SKU + Name */}
   607→                        <TableCell onClick={e => e.stopPropagation()}>
   608→                          <div>
   609→                            <p className="font-medium text-sm text-slate-800 truncate max-w-[200px]">{order.productName}</p>
   610→                            <p className="text-xs text-slate-500">{order.productSku}</p>
   611→                          </div>
   612→                        </TableCell>
   613→
   614→                        {/* Category */}
   615→                        <TableCell className="hidden lg:table-cell" onClick={e => e.stopPropagation()}>
   616→                          <span className="text-xs text-slate-600">{order.category || '—'}</span>
   617→                        </TableCell>
   618→
   619→                        {/* Stock / ROP */}
   620→                        <TableCell onClick={e => e.stopPropagation()}>
   621→                          <div className="flex items-center gap-1.5">
   622→                            <span className={`font-semibold text-sm ${stockColor(order.currentStock, order.reorderPoint)}`}>
   623→                              {order.currentStock}
   624→                            </span>
   625→                            <span className="text-slate-400 text-xs">/</span>
   626→                            <span className="text-xs text-slate-500">{order.reorderPoint}</span>
   627→                          </div>
   628→                        </TableCell>
   629→
   630→                        {/* Suggested Qty */}
   631→                        <TableCell onClick={e => e.stopPropagation()}>
   632→                          <span className="font-semibold text-sm text-slate-800">{order.suggestedQty}</span>
   633→                        </TableCell>
   634→
   635→                        {/* Order Trigger Date */}
   636→                        <TableCell className="hidden xl:table-cell" onClick={e => e.stopPropagation()}>
   637→                          <span className="text-xs text-slate-600">{formatDate(order.orderTriggerDate)}</span>
   638→                        </TableCell>
   639→
   640→                        {/* Expected Delivery Date */}
   641→                        <TableCell className="hidden xl:table-cell" onClick={e => e.stopPropagation()}>
   642→                          <span className="text-xs text-slate-600">{formatDate(order.expectedDeliveryDate)}</span>
   643→                        </TableCell>
   644→
   645→                        {/* Lead Time */}
   646→                        <TableCell className="hidden lg:table-cell" onClick={e => e.stopPropagation()}>
   647→                          <span className="text-xs text-slate-600">{order.totalLeadTime}d</span>
   648→                        </TableCell>
   649→
   650→                        {/* CNY Risk */}
   651→                        <TableCell className="hidden lg:table-cell" onClick={e => e.stopPropagation()}>
   652→                          {order.cnyRisk ? (
   653→                            <ShieldAlert className="h-4 w-4 text-red-500" />
   654→                          ) : (
   655→                            <span className="text-slate-300">—</span>
   656→                          )}
   657→                        </TableCell>
   658→
   659→                        {/* Season */}
   660→                        <TableCell className="hidden xl:table-cell" onClick={e => e.stopPropagation()}>
   661→                          <Badge variant="outline" className={`text-[10px] ${SEASON_COLORS[order.season] || 'bg-slate-100 text-slate-600'}`}>
   662→                            {SEASON_LABELS[order.season] || order.season}
   663→                          </Badge>
   664→                        </TableCell>
   665→
   666→                        {/* Actions */}
   667→                        <TableCell onClick={e => e.stopPropagation()}>
   668→                          {order.status === 'pending' ? (
   669→                            <Button
   670→                              size="sm"
   671→                              variant="outline"
   672→                              className="h-7 text-xs gap-1 border-emerald-200 text-emerald-700 hover:bg-emerald-50 hover:text-emerald-800"
   673→                              disabled={isConverting}
   674→                              onClick={(e: React.MouseEvent) => {
   675→                                e.stopPropagation();
   676→                                handleConvert(order);
   677→                              }}
   678→                            >
   679→                              {isConverting ? (
   680→                                <Loader2 className="h-3 w-3 animate-spin" />
   681→                              ) : (
   682→                                <ShoppingCart className="h-3 w-3" />
   683→                              )}
   684→                              Convert to PO
   685→                            </Button>
   686→                          ) : (
   687→                            <Badge variant="outline" className={`text-[10px] ${STATUS_BADGE[order.status]?.className || ''}`}>
   688→                              {STATUS_BADGE[order.status]?.label || order.status}
   689→                            </Badge>
   690→                          )}
   691→                        </TableCell>
   692→                      </motion.tr>
   693→                    );
   694→                  })}
   695→                </AnimatePresence>
   696→              </TableBody>
   697→            </Table>
   698→          </div>
   699→
   700→          {/* Expanded row detail (desktop) */}
   701→          <AnimatePresence>
   702→            {expandedRow && (() => {
   703→              const order = orders.find(o => o.id === expandedRow);
   704→              if (!order) return null;
   705→              return (
   706→                <motion.div
   707→                  key={`expanded-${order.id}`}
   708→                  initial={{ opacity: 0, height: 0 }}
   709→                  animate={{ opacity: 1, height: 'auto' }}
   710→                  exit={{ opacity: 0, height: 0 }}
   711→                  transition={{ duration: 0.2 }}
   712→                  className="overflow-hidden hidden md:block"
   713→                >
   714→                  <ExpandedOrderDetail order={order} />
   715→                </motion.div>
   716→              );
   717→            })()}
   718→          </AnimatePresence>
   719→
   720→          {/* ========== Mobile Card View ========== */}
   721→          <div className="md:hidden space-y-2">
   722→            <AnimatePresence initial={false}>
   723→              {paged.map(order => {
   724→                const pCfg = PRIORITY_CONFIG[order.priority] || PRIORITY_CONFIG.normal;
   725→                const isExpanded = expandedRow === order.id;
   726→                const isConverting = convertingId === order.id;
   727→
   728→                return (
   729→                  <motion.div
   730→                    key={order.id}
   731→                    initial={{ opacity: 0, y: -4 }}
   732→                    animate={{ opacity: 1, y: 0 }}
   733→                    exit={{ opacity: 0, y: 4 }}
   734→                    transition={{ duration: 0.15 }}
   735→                  >
   736→                    <Card
   737→                      className={`cursor-pointer shadow-sm ${pCfg.border} ${isExpanded ? 'ring-1 ring-slate-200' : ''}`}
   738→                      onClick={() => setExpandedRow(isExpanded ? null : order.id)}
   739→                    >
   740→                      <CardContent className="p-3 space-y-2">
   741→                        {/* Row 1: Priority + Product */}
   742→                        <div className="flex items-start justify-between gap-2">
   743→                          <div className="flex items-center gap-2 min-w-0">
   744→                            <Badge className={`${pCfg.bg} text-[10px]`}>{pCfg.label}</Badge>
   745→                            <div className="min-w-0">
   746→                              <p className="text-sm font-medium text-slate-800 truncate">{order.productName}</p>
   747→                              <p className="text-xs text-slate-500">{order.productSku}</p>
   748→                            </div>
   749→                          </div>
   750→                          {order.cnyRisk && <ShieldAlert className="h-4 w-4 text-red-500 flex-shrink-0" />}
   751→                        </div>
   752→
   753→                        {/* Row 2: Key numbers */}
   754→                        <div className="grid grid-cols-3 gap-2 text-center">
   755→                          <div className="p-1.5 rounded bg-slate-50">
   756→                            <p className={`text-sm font-bold ${stockColor(order.currentStock, order.reorderPoint)}`}>{order.currentStock}</p>
   757→                            <p className="text-[9px] text-slate-500">Stock</p>
   758→                          </div>
   759→                          <div className="p-1.5 rounded bg-slate-50">
   760→                            <p className="text-sm font-bold text-slate-700">{order.suggestedQty}</p>
   761→                            <p className="text-[9px] text-slate-500">Suggested</p>
   762→                          </div>
   763→                          <div className="p-1.5 rounded bg-slate-50">
   764→                            <p className="text-sm font-bold text-slate-700">{order.totalLeadTime}d</p>
   765→                            <p className="text-[9px] text-slate-500">Lead</p>
   766→                          </div>
   767→                        </div>
   768→
   769→                        {/* Row 3: Dates + Season */}
   770→                        <div className="flex items-center justify-between text-xs text-slate-500">
   771→                          <span>Order: {formatDate(order.orderTriggerDate)}</span>
   772→                          <Badge variant="outline" className={`text-[9px] ${SEASON_COLORS[order.season] || 'bg-slate-100 text-slate-600'}`}>
   773→                            {SEASON_LABELS[order.season] || order.season}
   774→                          </Badge>
   775→                        </div>
   776→
   777→                        <div className="text-xs text-slate-500">
   778→                          Deliver: {formatDate(order.expectedDeliveryDate)}
   779→                        </div>
   780→
   781→                        {/* Action */}
   782→                        {order.status === 'pending' ? (
   783→                          <Button
   784→                            size="sm"
   785→                            variant="outline"
   786→                            className="w-full h-7 text-xs gap-1 border-emerald-200 text-emerald-700 hover:bg-emerald-50"
   787→                            disabled={isConverting}
   788→                            onClick={e => { e.stopPropagation(); handleConvert(order); }}
   789→                          >
   790→                            {isConverting ? <Loader2 className="h-3 w-3 animate-spin" /> : <ShoppingCart className="h-3 w-3" />}
   791→                            Convert to PO
   792→                          </Button>
   793→                        ) : (
   794→                          <Badge variant="outline" className={`text-[10px] ${STATUS_BADGE[order.status]?.className || ''}`}>
   795→                            {STATUS_BADGE[order.status]?.label || order.status}
   796→                          </Badge>
   797→                        )}
   798→                      </CardContent>
   799→                    </Card>
   800→
   801→                    {/* Expanded detail on mobile */}
   802→                    <AnimatePresence>
   803→                      {isExpanded && (
   804→                        <motion.div
   805→                          initial={{ opacity: 0, height: 0 }}
   806→                          animate={{ opacity: 1, height: 'auto' }}
   807→                          exit={{ opacity: 0, height: 0 }}
   808→                          transition={{ duration: 0.2 }}
   809→                          className="overflow-hidden"
   810→                        >
   811→                          <ExpandedOrderDetail order={order} />
   812→                        </motion.div>
   813→                      )}
   814→                    </AnimatePresence>
   815→                  </motion.div>
   816→                );
   817→              })}
   818→            </AnimatePresence>
   819→          </div>
   820→
   821→          {/* ========== Pagination ========== */}
   822→          {totalPages > 1 && (
   823→            <div className="flex items-center justify-between pt-2">
   824→              <p className="text-xs text-slate-500">
   825→                Showing {(safePage - 1) * PAGE_SIZE + 1}–{Math.min(safePage * PAGE_SIZE, filtered.length)} of {filtered.length}
   826→              </p>
   827→              <Pagination>
   828→                <PaginationContent>
   829→                  <PaginationItem>
   830→                    <PaginationPrevious
   831→                      onClick={() => setPage(p => Math.max(1, p - 1))}
   832→                      className={safePage <= 1 ? 'pointer-events-none opacity-50' : 'cursor-pointer'}
   833→                    />
   834→                  </PaginationItem>
   835→                  {paginationRange.map((item, i) =>
   836→                    item === 'ellipsis' ? (
   837→                      <PaginationItem key={`ellipsis-${i}`}>
   838→                        <PaginationEllipsis />
   839→                      </PaginationItem>
   840→                    ) : (
   841→                      <PaginationItem key={item}>
   842→                        <PaginationLink
   843→                          isActive={safePage === item}
   844→                          onClick={() => setPage(item)}
   845→                          className="cursor-pointer"
   846→                        >
   847→                          {item}
   848→                        </PaginationLink>
   849→                      </PaginationItem>
   850→                    )
   851→                  )}
   852→                  <PaginationItem>
   853→                    <PaginationNext
   854→                      onClick={() => setPage(p => Math.min(totalPages, p + 1))}
   855→                      className={safePage >= totalPages ? 'pointer-events-none opacity-50' : 'cursor-pointer'}
   856→                    />
   857→                  </PaginationItem>
   858→                </PaginationContent>
   859→              </Pagination>
   860→            </div>
   861→          )}
   862→        </>
   863→      )}
   864→    </div>
   865→  );
   866→}
   867→

   869→export type { RecommendedOrderRow };
