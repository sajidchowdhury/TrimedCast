    67→function ValidateStep() {
    68→  const { currentImportId, setLoading, setError, setValidationResult, setStep, isLoading } = useETLStore();
    69→  const handleProceed = useCallback(() => { setStep(3); }, [setStep]);
    70→
    71→  useEffect(() => {
    72→    const runValidation = async () => {
    73→      if (!currentImportId) return;
    74→      setLoading(true);
    75→      setError(null);
    76→      try {
    77→        const res = await fetch(`/api/imports/${currentImportId}/validate`, { method: 'POST' });
    78→        if (!res.ok) { const err = await res.json().catch(() => ({})); throw new Error(err.error || 'Validation failed'); }
    79→        const json = await res.json();
    80→        if (!json.success) throw new Error(json.error || 'Validation failed');
    81→        setValidationResult(json.data as ValidationResultClient);
    82→      } catch (err) { setError(err instanceof Error ? err.message : 'Validation failed'); }
    83→      finally { setLoading(false); }
    84→    };
    85→    runValidation();
    86→  }, [currentImportId, setLoading, setError, setValidationResult]);
    87→
    88→  return (
    89→    <div>
    90→      {isLoading && (
    91→        <Card className="mb-4">
    92→          <CardContent className="py-8 text-center">
    93→            <Loader2 className="h-8 w-8 mx-auto text-blue-500 animate-spin mb-3" />
    94→            <p className="text-sm text-gray-600">Running 3-phase validation...</p>
    95→            <p className="text-xs text-gray-400 mt-1">Schema → Data → Business Rules</p>
    96→          </CardContent>
    97→        </Card>
    98→      )}
    99→      <ValidationResults onProceed={handleProceed} />
   100→    </div>
   101→  );
   102→}

   104→function HarmonizeStep() {
   105→  const { currentImportId, setLoading, setError, setHarmonizationResult, setStep, harmonizationResult, isLoading } = useETLStore();
   106→  const handleProceed = useCallback(() => { setStep(4); }, [setStep]);
   107→
   108→  useEffect(() => {
   109→    if (harmonizationResult) return;
   110→    const runHarmonization = async () => {
   111→      if (!currentImportId) return;
   112→      setLoading(true); setError(null);
   113→      try {
   114→        const res = await fetch(`/api/imports/${currentImportId}/harmonize`, { method: 'POST' });
   115→        if (!res.ok) { const err = await res.json().catch(() => ({})); throw new Error(err.error || 'Harmonization failed'); }
   116→        const json = await res.json();
   117→        if (!json.success) throw new Error(json.error || 'Harmonization failed');
   118→        setHarmonizationResult(json.data as HarmonizationResultClient);
   119→      } catch (err) { setError(err instanceof Error ? err.message : 'Harmonization failed'); }
   120→      finally { setLoading(false); }
   121→    };
   122→    runHarmonization();
   123→  }, [currentImportId, harmonizationResult, setLoading, setError, setHarmonizationResult]);
   124→
   125→  if (isLoading && !harmonizationResult) {
   126→    return (
   127→      <Card>
   128→        <CardContent className="py-8 text-center">
   129→          <Loader2 className="h-8 w-8 mx-auto text-blue-500 animate-spin mb-3" />
   130→          <p className="text-sm text-gray-600">Running 6-step harmonization...</p>
   131→          <p className="text-xs text-gray-400 mt-1">Trim → Dates → Categories → Units → Dedup → Enrich</p>
   132→        </CardContent>
   133→      </Card>
   134→    );
   135→  }
   136→  return <HarmonizationLog onProceed={handleProceed} />;
   137→}

   139→function InsertStep() {
   140→  const { currentImportId, setLoading, setError, setInsertionResult, setStep, insertionResult, uploadResult, mappings, importType, isLoading } = useETLStore();
   141→
   142→  useEffect(() => {
   143→    if (insertionResult) return;
   144→    const runInsert = async () => {
   145→      if (!currentImportId) return;
   146→      setLoading(true); setError(null);
   147→      try {
   148→        const res = await fetch(`/api/imports/${currentImportId}/insert`, { method: 'POST' });
   149→        if (!res.ok) { const err = await res.json().catch(() => ({})); throw new Error(err.error || 'Insertion failed'); }
   150→        const json = await res.json();
   151→        if (!json.success) throw new Error(json.error || 'Insertion failed');
   152→        setInsertionResult(json.data as InsertionResultClient);
   153→      } catch (err) { setError(err instanceof Error ? err.message : 'Insertion failed'); }
   154→      finally { setLoading(false); }
   155→    };
   156→    runInsert();
   157→  }, [currentImportId, insertionResult, setLoading, setError, setInsertionResult]);
   158→
   159→  if (isLoading && !insertionResult) {
   160→    return (
   161→      <Card>
   162→        <CardContent className="py-8 text-center">
   163→          <Loader2 className="h-8 w-8 mx-auto text-blue-500 animate-spin mb-3" />
   164→          <p className="text-sm text-gray-600">Inserting data into database...</p>
   165→        </CardContent>
   166→      </Card>
   167→    );
   168→  }
   169→  if (!insertionResult) return null;
   170→
   171→  const schema = importType ? IMPORT_TYPE_SCHEMAS[importType as ImportType] : null;
   172→  const requiredTotal = schema?.requiredFields.length || 0;
   173→  const requiredMapped = mappings.filter((m) => m.targetField && m.confidence > 0 && schema?.requiredFields.some((rf) => rf.field === m.targetField)).length;
   174→  const qualityStats: QualityStats = {
   175→    rowsTotal: uploadResult?.rowsTotal || 0,
   176→    rowsValid: uploadResult?.rowsTotal || insertionResult.inserted,
   177→    rowsInserted: insertionResult.inserted,
   178→    rowsDuplicate: 0,
   179→    requiredMapped,
   180→    requiredTotal,
   181→  };
   182→
   183→  return (
   184→    <motion.div initial={{ opacity: 0, y: 20 }} animate={{ opacity: 1, y: 0 }} transition={{ duration: 0.4 }} className="space-y-4">
   185→      <Card className="border-emerald-200 bg-emerald-50/50">
   186→        <CardContent className="py-6">
   187→          <div className="flex items-center gap-3 mb-4">
   188→            <div className="w-12 h-12 rounded-full bg-emerald-100 flex items-center justify-center">
   189→              <CheckCircle2 className="h-7 w-7 text-emerald-600" />
   190→            </div>
   191→            <div>
   192→              <h2 className="text-xl font-bold text-emerald-800">Import Complete!</h2>
   193→              <p className="text-sm text-emerald-600">Your data has been successfully processed and inserted</p>
   194→            </div>
   195→          </div>
   196→          <QualityBadge score={insertionResult.qualityScore} size="lg" showBreakdown={true} stats={qualityStats} />
   197→        </CardContent>
   198→      </Card>
   199→      <div className="grid grid-cols-2 sm:grid-cols-4 gap-3">
   200→        <Card><CardContent className="pt-4 pb-3 text-center"><Database className="h-5 w-5 mx-auto text-blue-500 mb-1" /><div className="text-2xl font-bold text-gray-700">{insertionResult.inserted}</div><div className="text-xs text-gray-500">Rows Inserted</div></CardContent></Card>
   201→        <Card><CardContent className="pt-4 pb-3 text-center"><XCircle className="h-5 w-5 mx-auto text-red-500 mb-1" /><div className="text-2xl font-bold text-gray-700">{insertionResult.skipped}</div><div className="text-xs text-gray-500">Rows Skipped</div></CardContent></Card>
   202→        <Card><CardContent className="pt-4 pb-3 text-center"><FileSpreadsheet className="h-5 w-5 mx-auto text-gray-500 mb-1" /><div className="text-2xl font-bold text-gray-700">{insertionResult.createdMasterData?.length || 0}</div><div className="text-xs text-gray-500">Master Data Created</div></CardContent></Card>
   203→        <Card><CardContent className="pt-4 pb-3 text-center"><Loader2 className="h-5 w-5 mx-auto text-gray-400 mb-1" style={{ animation: 'none' }} /><div className="text-2xl font-bold text-gray-700">{insertionResult.durationMs !== null ? insertionResult.durationMs < 1000 ? `${insertionResult.durationMs}ms` : `${(insertionResult.durationMs / 1000).toFixed(1)}s` : '—'}</div><div className="text-xs text-gray-500">Duration</div></CardContent></Card>
   204→      </div>
   205→      {insertionResult.createdMasterData && insertionResult.createdMasterData.length > 0 && (
   206→        <Card>
   207→          <CardHeader className="pb-2"><CardTitle className="text-sm font-medium text-gray-600">Auto-Created Master Data</CardTitle></CardHeader>
   208→          <CardContent><ul className="text-xs text-gray-500 space-y-1">{insertionResult.createdMasterData.map((item, i) => (<li key={i} className="flex items-center gap-2"><Badge variant="outline" className="text-xs h-5">New</Badge>{item}</li>))}</ul></CardContent>
   209→        </Card>
   210→      )}
   211→      {insertionResult.errors && insertionResult.errors.length > 0 && (
   212→        <Card>
   213→          <CardHeader className="pb-2"><CardTitle className="text-sm font-medium text-red-600">Insertion Errors ({insertionResult.errors.length})</CardTitle></CardHeader>
   214→          <CardContent><ul className="text-xs text-red-500 space-y-1">{insertionResult.errors.slice(0, 10).map((err, i) => (<li key={i}>Row {err.row}: {err.error}</li>))}{insertionResult.errors.length > 10 && <li className="text-gray-400">...and {insertionResult.errors.length - 10} more</li>}</ul></CardContent>
   215→        </Card>
   216→      )}
   217→    </motion.div>
   218→  );
   219→}
