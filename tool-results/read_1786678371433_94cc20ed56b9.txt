    66→function ValidateStep() {
    67→  const { currentImportId, setLoading, setError, setValidationResult, setStep, isLoading } = useETLStore();
    68→  const handleProceed = useCallback(() => { setStep(3); }, [setStep]);
    69→
    70→  useEffect(() => {
    71→    const runValidation = async () => {
    72→      if (!currentImportId) return;
    73→      setLoading(true);
    74→      setError(null);
    75→      try {
    76→        const res = await fetch(`/api/imports/${currentImportId}/validate`, { method: 'POST' });
    77→        if (!res.ok) { const err = await res.json().catch(() => ({})); throw new Error(err.error || 'Validation failed'); }
    78→        const json = await res.json();
    79→        if (!json.success) throw new Error(json.error || 'Validation failed');
    80→        setValidationResult(json.data as ValidationResultClient);
    81→      } catch (err) { setError(err instanceof Error ? err.message : 'Validation failed'); }
    82→      finally { setLoading(false); }
    83→    };
    84→    runValidation();
    85→  }, [currentImportId, setLoading, setError, setValidationResult]);
    86→
    87→  return (
    88→    <div>
    89→      {isLoading && (
    90→        <Card className="mb-4">
    91→          <CardContent className="py-8 text-center">
    92→            <Loader2 className="h-8 w-8 mx-auto text-blue-500 animate-spin mb-3" />
    93→            <p className="text-sm text-gray-600">Running 3-phase validation...</p>
    94→            <p className="text-xs text-gray-400 mt-1">Schema → Data → Business Rules</p>
    95→          </CardContent>
    96→        </Card>
    97→      )}
    98→      <ValidationResults onProceed={handleProceed} />
    99→    </div>
   100→  );
   101→}

   103→function HarmonizeStep() {
   104→  const { currentImportId, setLoading, setError, setHarmonizationResult, setStep, harmonizationResult, isLoading } = useETLStore();
   105→  const handleProceed = useCallback(() => { setStep(4); }, [setStep]);
   106→
   107→  useEffect(() => {
   108→    if (harmonizationResult) return;
   109→    const runHarmonization = async () => {
   110→      if (!currentImportId) return;
   111→      setLoading(true); setError(null);
   112→      try {
   113→        const res = await fetch(`/api/imports/${currentImportId}/harmonize`, { method: 'POST' });
   114→        if (!res.ok) { const err = await res.json().catch(() => ({})); throw new Error(err.error || 'Harmonization failed'); }
   115→        const json = await res.json();
   116→        if (!json.success) throw new Error(json.error || 'Harmonization failed');
   117→        setHarmonizationResult(json.data as HarmonizationResultClient);
   118→      } catch (err) { setError(err instanceof Error ? err.message : 'Harmonization failed'); }
   119→      finally { setLoading(false); }
   120→    };
   121→    runHarmonization();
   122→  }, [currentImportId, harmonizationResult, setLoading, setError, setHarmonizationResult]);
   123→
   124→  if (isLoading && !harmonizationResult) {
   125→    return (
   126→      <Card>
   127→        <CardContent className="py-8 text-center">
   128→          <Loader2 className="h-8 w-8 mx-auto text-blue-500 animate-spin mb-3" />
   129→          <p className="text-sm text-gray-600">Running 6-step harmonization...</p>
   130→          <p className="text-xs text-gray-400 mt-1">Trim → Dates → Categories → Units → Dedup → Enrich</p>
   131→        </CardContent>
   132→      </Card>
   133→    );
   134→  }
   135→  return <HarmonizationLog onProceed={handleProceed} />;
   136→}

   138→function InsertStep() {
   139→  const { currentImportId, setLoading, setError, setInsertionResult, setStep, insertionResult, uploadResult, mappings, importType, isLoading } = useETLStore();
   140→
   141→  useEffect(() => {
   142→    if (insertionResult) return;
   143→    const runInsert = async () => {
   144→      if (!currentImportId) return;
   145→      setLoading(true); setError(null);
   146→      try {
   147→        const res = await fetch(`/api/imports/${currentImportId}/insert`, { method: 'POST' });
   148→        if (!res.ok) { const err = await res.json().catch(() => ({})); throw new Error(err.error || 'Insertion failed'); }
   149→        const json = await res.json();
   150→        if (!json.success) throw new Error(json.error || 'Insertion failed');
   151→        setInsertionResult(json.data as InsertionResultClient);
   152→      } catch (err) { setError(err instanceof Error ? err.message : 'Insertion failed'); }
   153→      finally { setLoading(false); }
   154→    };
   155→    runInsert();
   156→  }, [currentImportId, insertionResult, setLoading, setError, setInsertionResult]);
   157→
   158→  if (isLoading && !insertionResult) {
   159→    return (
   160→      <Card>
   161→        <CardContent className="py-8 text-center">
   162→          <Loader2 className="h-8 w-8 mx-auto text-blue-500 animate-spin mb-3" />
   163→          <p className="text-sm text-gray-600">Inserting data into database...</p>
   164→        </CardContent>
   165→      </Card>
   166→    );
   167→  }
   168→  if (!insertionResult) return null;
   169→
   170→  const schema = importType ? IMPORT_TYPE_SCHEMAS[importType as ImportType] : null;
   171→  const requiredTotal = schema?.requiredFields.length || 0;
   172→  const requiredMapped = mappings.filter((m) => m.targetField && m.confidence > 0 && schema?.requiredFields.some((rf) => rf.field === m.targetField)).length;
   173→  const qualityStats: QualityStats = {
   174→    rowsTotal: uploadResult?.rowsTotal || 0,
   175→    rowsValid: uploadResult?.rowsTotal || insertionResult.inserted,
   176→    rowsInserted: insertionResult.inserted,
   177→    rowsDuplicate: 0,
   178→    requiredMapped,
   179→    requiredTotal,
   180→  };
   181→
   182→  return (
   183→    <motion.div initial={{ opacity: 0, y: 20 }} animate={{ opacity: 1, y: 0 }} transition={{ duration: 0.4 }} className="space-y-4">
   184→      <Card className="border-emerald-200 bg-emerald-50/50">
   185→        <CardContent className="py-6">
   186→          <div className="flex items-center gap-3 mb-4">
   187→            <div className="w-12 h-12 rounded-full bg-emerald-100 flex items-center justify-center">
   188→              <CheckCircle2 className="h-7 w-7 text-emerald-600" />
   189→            </div>
   190→            <div>
   191→              <h2 className="text-xl font-bold text-emerald-800">Import Complete!</h2>
   192→              <p className="text-sm text-emerald-600">Your data has been successfully processed and inserted</p>
   193→            </div>
   194→          </div>
   195→          <QualityBadge score={insertionResult.qualityScore} size="lg" showBreakdown={true} stats={qualityStats} />
   196→        </CardContent>
   197→      </Card>
   198→      <div className="grid grid-cols-2 sm:grid-cols-4 gap-3">
   199→        <Card><CardContent className="pt-4 pb-3 text-center"><Database className="h-5 w-5 mx-auto text-blue-500 mb-1" /><div className="text-2xl font-bold text-gray-700">{insertionResult.inserted}</div><div className="text-xs text-gray-500">Rows Inserted</div></CardContent></Card>
   200→        <Card><CardContent className="pt-4 pb-3 text-center"><XCircle className="h-5 w-5 mx-auto text-red-500 mb-1" /><div className="text-2xl font-bold text-gray-700">{insertionResult.skipped}</div><div className="text-xs text-gray-500">Rows Skipped</div></CardContent></Card>
   201→        <Card><CardContent className="pt-4 pb-3 text-center"><FileSpreadsheet className="h-5 w-5 mx-auto text-gray-500 mb-1" /><div className="text-2xl font-bold text-gray-700">{insertionResult.createdMasterData?.length || 0}</div><div className="text-xs text-gray-500">Master Data Created</div></CardContent></Card>
   202→        <Card><CardContent className="pt-4 pb-3 text-center"><Loader2 className="h-5 w-5 mx-auto text-gray-400 mb-1" style={{ animation: 'none' }} /><div className="text-2xl font-bold text-gray-700">{insertionResult.durationMs !== null ? insertionResult.durationMs < 1000 ? `${insertionResult.durationMs}ms` : `${(insertionResult.durationMs / 1000).toFixed(1)}s` : '—'}</div><div className="text-xs text-gray-500">Duration</div></CardContent></Card>
   203→      </div>
   204→      {insertionResult.createdMasterData && insertionResult.createdMasterData.length > 0 && (
   205→        <Card>
   206→          <CardHeader className="pb-2"><CardTitle className="text-sm font-medium text-gray-600">Auto-Created Master Data</CardTitle></CardHeader>
   207→          <CardContent><ul className="text-xs text-gray-500 space-y-1">{insertionResult.createdMasterData.map((item, i) => (<li key={i} className="flex items-center gap-2"><Badge variant="outline" className="text-xs h-5">New</Badge>{item}</li>))}</ul></CardContent>
   208→        </Card>
   209→      )}
   210→      {insertionResult.errors && insertionResult.errors.length > 0 && (
   211→        <Card>
   212→          <CardHeader className="pb-2"><CardTitle className="text-sm font-medium text-red-600">Insertion Errors ({insertionResult.errors.length})</CardTitle></CardHeader>
   213→          <CardContent><ul className="text-xs text-red-500 space-y-1">{insertionResult.errors.slice(0, 10).map((err, i) => (<li key={i}>Row {err.row}: {err.error}</li>))}{insertionResult.errors.length > 10 && <li className="text-gray-400">...and {insertionResult.errors.length - 10} more</li>}</ul></CardContent>
   214→        </Card>
   215→      )}
   216→    </motion.div>
   217→  );
   218→}
