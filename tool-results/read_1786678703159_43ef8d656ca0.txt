   109→function formatDate(dateStr: string): string {
   110→  try {
   111→    const d = new Date(dateStr);
   112→    return d.toLocaleDateString('en-US', { month: 'short', day: 'numeric' });
   113→  } catch {
   114→    return dateStr;
   115→  }
   116→}
   117→
   118→function formatBDT(amount: number): string {
   119→  return '৳' + amount.toLocaleString('en-BD', { minimumFractionDigits: 0, maximumFractionDigits: 0 });
   120→}
   121→
   122→function stockColor(current: number, reorder: number): string {
   123→  if (current <= 0) return 'text-red-600';
   124→  if (current < reorder * 0.5) return 'text-red-500';
   125→  if (current < reorder) return 'text-amber-500';
   126→  return 'text-emerald-500';
   127→}

   170→            productSku: (t.productSku as string) || '',

   198→  useEffect(() => { fetchOrders(); }, [fetchOrders]);

   204→      const res = await fetch('/api/orders/convert', {
   205→        method: 'POST',
   206→        headers: { 'Content-Type': 'application/json' },

   209→          recommendedOrderId: order.id,

   213→        const json = await res.json();

   234→      }

   236→      return field;

   279→          cmp = a.orderTriggerDate.localeCompare(b.orderTriggerDate);
   280→          break;

   282→          cmp = a.suggestedQty - b.suggestedQty;
   283→          break;
   284→        case 'totalLeadTime':
   285→          cmp = a.totalLeadTime - b.totalLeadTime;
   286→          break;
   287→      }
   288→      return sortDir === 'asc' ? cmp : -cmp;
   289→    });

   310→      if (safePage < totalPages - 2) range.push('ellipsis');

   314→  }, [totalPages, safePage]);

   329→        className={`h-3 w-3 inline ml-1 ${isActive ? 'text-emerald-600' : 'text-slate-400'}`}

   343→          <CardContent className="p-3 flex items-center gap-3">
   344→            <div className="w-9 h-9 rounded-lg bg-slate-100 flex items-center justify-center">

   348→              <p className="text-[10px] uppercase tracking-wider text-slate-500">Total Orders</p>

   350→            </div>
   351→          </CardContent>
   352→        </Card>

   354→        <Card className="bg-white shadow-sm">

   360→              <p className="text-[10px] uppercase tracking-wider text-slate-500">Urgent</p>
   361→              <p className="text-lg font-bold text-red-600">{filteredSummary.totalUrgent}</p>
   362→            </div>
   363→          </CardContent>
   364→        </Card>
   365→
   366→        <Card className="bg-white shadow-sm">
   367→          <CardContent className="p-3 flex items-center gap-3">
   368→            <div className="w-9 h-9 rounded-lg bg-orange-50 flex items-center justify-center">

   370→            </div>

   394→            <div className="w-9 h-9 rounded-lg bg-red-50 flex items-center justify-center">
   395→              <ShieldAlert className="h-4 w-4 text-red-500" />
   396→            </div>
   397→            <div>
   398→              <p className="text-[10px] uppercase tracking-wider text-slate-500">CNY Risk</p>

   405→      {/* ========== Filter Bar ========== */}

   407→        <Select value={priorityFilter} onValueChange={setPriorityFilter}>
   408→          <SelectTrigger size="sm" className="w-[130px]">

   411→          <SelectContent>

   533→                <AnimatePresence initial={false}>
   534→                  {paged.map(order => {
