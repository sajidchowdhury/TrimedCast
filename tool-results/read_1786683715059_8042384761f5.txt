   107→   107→  customsSeaWorst: number;   // 18 days
   108→   108→
   109→   109→  // Customs (Air/Dhaka)
   110→   110→  customsAirBest: number;    // 2 days
   111→   111→  customsAirLikely: number;  // 3 days
   112→   112→  customsAirWorst: number;   // 5 days
   113→   113→}
   114→   114→
   115→   115→export const DEFAULT_LT_DISTRIBUTION: LeadTimeDistribution = {
   116→   116→  mfgBest: 75, mfgLikely: 90, mfgWorst: 120,
   117→   117→  shipSeaBest: 45, shipSeaLikely: 52, shipSeaWorst: 65,
   118→   118→  shipAirBest: 5, shipAirLikely: 8, shipAirWorst: 12,
   119→   119→  customsSeaBest: 7, customsSeaLikely: 10, customsSeaWorst: 18,
   120→   120→  customsAirBest: 2, customsAirLikely: 3, customsAirWorst: 5,
   121→   121→};
   122→   122→
   123→   123→/** PERT expected value: (best + 4*likely + worst) / 6 */
   124→   124→function pertExpected(best: number, likely: number, worst: number): number {
   125→   125→  return (best + 4 * likely + worst) / 6;
   126→   126→}
   127→   127→
   128→   128→/** Approximate P90 using PERT: mean + 1.28 * stddev */
   129→   129→function pertP90(best: number, likely: number, worst: number): number {
   130→   130→  const mean = pertExpected(best, likely, worst);
   131→   131→  const stddev = (worst - best) / 6;
   132→   132→  return mean + 1.28 * stddev;
   133→   133→}
   134→   134→
   135→   135→export function calculateLeadTimeP90(
   136→   136→  dist: LeadTimeDistribution = DEFAULT_LT_DISTRIBUTION,
   137→   137→  mode: 'sea' | 'air' = 'sea'
   138→   138→): { totalP90: number; mfgP90: number; shipP90: number; customsP90: number } {
   139→   139→  const mfgP90 = pertP90(dist.mfgBest, dist.mfgLikely, dist.mfgWorst);
   140→   140→  const shipP90 = mode === 'sea'
   141→   141→    ? pertP90(dist.shipSeaBest, dist.shipSeaLikely, dist.shipSeaWorst)
   142→   142→    : pertP90(dist.shipAirBest, dist.shipAirLikely, dist.shipAirWorst);
   143→   143→  const customsP90 = mode === 'sea'
   144→   144→    ? pertP90(dist.customsSeaBest, dist.customsSeaLikely, dist.customsSeaWorst)
   145→   145→    : pertP90(dist.customsAirBest, dist.customsAirLikely, dist.customsAirWorst);
   146→   146→
   147→   147→  return { totalP90: Math.round(mfgP90 + shipP90 + customsP90), mfgP90: Math.round(mfgP90), shipP90: Math.round(shipP90), customsP90: Math.round(customsP90) };
   148→   148→}

   228→   228→
   229→   229→export interface CNYRiskAssessment {
   230→   230→  hasRisk: boolean;
   231→   231→  overlapDays: number;
   232→   232→  effectiveCnyStart: Date;
   233→   233→  cnyShutdownStart: Date;
   234→   234→  cnyShutdownEnd: Date;
   235→   235→  strategy: CNYStrategy;
   236→   236→  additionalDelayDays: number;
   237→   237→  latestSafeOrderDate: Date | null;
   238→   238→  postCnyOrderDate: Date | null;
   239→   239→  explanation: string;
   240→   240→}
   241→   241→
   242→   242→function assessCNYRisk(
   243→   243→  orderTriggerDate: Date,
   244→   244→  mfgDays: number,
   245→   245→  cnyInfo: CNYInfo | null,
   246→   246→  today: Date,
   247→   247→): CNYRiskAssessment {
   248→   248→  if (!cnyInfo) {
   249→   249→    return {
   250→   250→      hasRisk: false, overlapDays: 0,
   251→   251→      effectiveCnyStart: today, cnyShutdownStart: today, cnyShutdownEnd: today,
   252→   252→      strategy: 'none', additionalDelayDays: 0,
   253→   253→      latestSafeOrderDate: null, postCnyOrderDate: null,
   254→   254→      explanation: 'No CNY window provided for analysis.',
   255→   255→    };
   256→   256→  }
   257→   257→
   258→   258→  const effectiveCnyStart = cnyInfo.effectiveStart;
   259→   259→  const effectiveCnyEnd = cnyInfo.effectiveEnd;
   260→   260→
   261→   261→  // Manufacturing period
   262→   262→  const orderProcDays = 2;
   263→   263→  const mfgStart = addDays(orderTriggerDate, orderProcDays);
   264→   264→  const mfgEnd = addDays(mfgStart, mfgDays);
   265→   265→
   266→   266→  // Check overlap
   267→   267→  const overlapStart = new Date(Math.max(mfgStart.getTime(), effectiveCnyStart.getTime()));
   268→   268→  const overlapEnd = new Date(Math.min(mfgEnd.getTime(), effectiveCnyEnd.getTime()));
   269→   269→
   270→   270→  if (overlapStart < overlapEnd) {
   271→   271→    const overlapDays = Math.round((overlapEnd.getTime() - overlapStart.getTime()) / (1000 * 60 * 60 * 24));
   272→   272→
   273→   273→    // Strategy A: Latest safe order date
   274→   274→    const latestSafe = addDays(effectiveCnyStart, -(mfgDays + orderProcDays));
   275→   275→
   276→   276→    // Strategy B: Post-CNY order date
   277→   277→    const postCnyOrder = effectiveCnyEnd;
   278→   278→
   279→   279→    // Total CNY shutdown period
   280→   280→    const cnyTotalDays = Math.round((effectiveCnyEnd.getTime() - effectiveCnyStart.getTime()) / (1000 * 60 * 60 * 24));
   281→   281→
   282→   282→    if (latestSafe >= today) {
   283→   283→      return {
   284→   284→        hasRisk: true, overlapDays,
   285→   285→        effectiveCnyStart, cnyShutdownStart: cnyInfo.startDate, cnyShutdownEnd: cnyInfo.endDate,
   286→   286→        strategy: 'before_cny', additionalDelayDays: 0,
   287→   287→        latestSafeOrderDate: latestSafe, postCnyOrderDate: null,
   288→   288→        explanation: `CNY RISK: Mfg overlaps CNY by ${overlapDays} days. STRATEGY A: Order by ${formatDate(latestSafe)} to complete manufacturing before CNY shutdown (${formatDate(cnyInfo.startDate)}–${formatDate(cnyInfo.endDate)}).`,
   289→   289→      };
   290→   290→    } else {
   291→   291→      return {
   292→   292→        hasRisk: true, overlapDays,
   293→   293→        effectiveCnyStart, cnyShutdownStart: cnyInfo.startDate, cnyShutdownEnd: cnyInfo.endDate,
   294→   294→        strategy: 'after_cny', additionalDelayDays: cnyTotalDays,
   295→   295→        latestSafeOrderDate: null, postCnyOrderDate: postCnyOrder,
   296→   296→        explanation: `CNY RISK: Mfg overlaps CNY by ${overlapDays} days. Cannot finish before CNY. STRATEGY B: Order after CNY on ${formatDate(postCnyOrder)}. Effective delay: ${cnyTotalDays} days.`,
   297→   297→      };
   298→   298→    }
   299→   299→  }
   300→   300→
   301→   301→  return {
   302→   302→    hasRisk: false, overlapDays: 0,
   303→   303→    effectiveCnyStart, cnyShutdownStart: cnyInfo.startDate, cnyShutdownEnd: cnyInfo.endDate,
   304→   304→    strategy: 'none', additionalDelayDays: 0,
   305→   305→    latestSafeOrderDate: null, postCnyOrderDate: null,
   306→   306→    explanation: `No CNY risk. Manufacturing completes before CNY window (${formatDate(cnyInfo.startDate)}–${formatDate(cnyInfo.endDate)}).`,
   307→   307→  };
   308→   308→}

   314→   314→export function selectCNYStrategy(
   315→   315→  daysUntilStockout: number,
   316→   316→  cnyDelayDays: number,
   317→   317→  itemMarginPct: number,
   318→   318→  itemUrgency: 'critical' | 'high' | 'normal' | 'low',
   319→   319→  canAirShip: boolean,
   320→   320→  airCostMultiplier: number = 8.0,
   321→   321→): CNYStrategy {
   322→   322→  // Can we survive the CNY delay with current stock?
   323→   323→  if (daysUntilStockout > cnyDelayDays + 30) {
   324→   324→    return 'after_cny';
   325→   325→  }
   326→   326→
   327→   327→  // Stockout risk — try to order before CNY
   328→   328→  if (itemUrgency === 'critical' || itemUrgency === 'high') {
   329→   329→    return 'before_cny';
   330→   330→  }
   331→   331→
   332→   332→  // Emergency: stockout imminent, consider air
   333→   333→  if (daysUntilStockout <= 30 && canAirShip && itemMarginPct >= 30) {
   334→   334→    return 'air_escape';
   335→   335→  }
   336→   336→
   337→   337→  // Default: partial approach
   338→   338→  return 'partial_order';
   339→   339→}

   427→   427→  constraintsApplied: string[];

   430→   430→
   431→   431→export function calculateRecommendedQty(
   432→   432→  forecastedDemand: number,
   433→   433→  safetyStock: number,
   434→   434→  currentStock: number,
   435→   435→  qtyOnOrder: number,
   436→   436→  eoq: number,
   437→   437→  moq: number,
   438→   438→  maxStock: number,
   439→   439→): QuantityBreakdown {
   440→   440→  const constraintsApplied: string[] = [];
   441→   441→
   442→   442→  // Step 1-3: Calculate demand gap
   443→   443→  const totalNeeded = forecastedDemand + safetyStock;
   444→   444→  const totalSupply = currentStock + qtyOnOrder;
   445→   445→  const gap = totalNeeded - totalSupply;
   446→   446→
   447→   447→  // Step 4: Check if stock is adequate
   448→   448→  if (gap <= 0) {
   449→   449→    return {
   450→   450→      recommendedQty: 0, status: 'adequate',
   451→   451→      reason: `Stock sufficient. Supply (${totalSupply}) covers need (${totalNeeded}).`,
   452→   452→      gap, eoq, moq, maxStock, constraintsApplied: [],
   453→   453→      totalNeeded, totalSupply, preConstraintQty: 0,
   454→   454→    };
   455→   455→  }
   456→   456→
   457→   457→  // Step 5: Check overstock
   458→   458→  if (currentStock > maxStock) {
   459→   459→    return {
   460→   460→      recommendedQty: 0, status: 'overstock',
   461→   461→      reason: `Currently overstocked (${currentStock} > max ${maxStock}). Allow draw-down.`,
   462→   462→      gap, eoq, moq, maxStock, constraintsApplied: ['overstock_drawdown'],
   463→   463→      totalNeeded, totalSupply, preConstraintQty: 0,
   464→   464→    };
   465→   465→  }
   466→   466→
   467→   467→  // Step 6: Start with economic quantity
   468→   468→  let qty = Math.max(gap, eoq);
   469→   469→  if (eoq > gap) {
   470→   470→    constraintsApplied.push(`eoq_floor: EOQ (${eoq}) > gap (${gap}), using EOQ`);
   471→   471→  }
   472→   472→
   473→   473→  // Step 7: Apply MOQ constraint
   474→   474→  if (qty < moq) {
   475→   475→    qty = moq;
   476→   476→    constraintsApplied.push(`moq_floor: qty raised to MOQ (${moq})`);
   477→   477→  }
   478→   478→
   479→   479→  const preConstraintQty = qty;
   480→   480→
   481→   481→  // Step 8: Apply warehouse capacity constraint
   482→   482→  const warehouseCapacityRemaining = maxStock - currentStock;
   483→   483→  if (qty > warehouseCapacityRemaining) {
   484→   484→    qty = warehouseCapacityRemaining;
   485→   485→    constraintsApplied.push(
   486→   486→      `warehouse_cap: qty reduced from ${preConstraintQty} to ${qty} (max_stock ${maxStock} - current ${currentStock})`
   487→   487→    );
   488→   488→  }
   489→   489→
   490→   490→  // Step 9: Check if warehouse can't even fit MOQ
   491→   491→  if (qty < moq && qty > 0) {
   492→   492→    return {
   493→   493→      recommendedQty: 0, status: 'warehouse_full',
   494→   494→      reason: `Insufficient warehouse space for MOQ. Available space: ${warehouseCapacityRemaining}, MOQ: ${moq}.`,
   495→   495→      gap, eoq, moq, maxStock, constraintsApplied: [...constraintsApplied, 'warehouse_below_moq'],
   496→   496→      totalNeeded, totalSupply, preConstraintQty,
   497→   497→    };
   498→   498→  }
   499→   499→
   500→   500→  qty = Math.max(0, qty);
   501→   501→  const status = qty > 0 ? 'order_needed' as const : 'adequate' as const;
   502→   502→  const reason = `Order ${qty} units. Gap: ${gap}, constrained by: ${constraintsApplied.length > 0 ? constraintsApplied.join('; ') : 'none'}`;
   503→   503→
   504→   504→  return {
   505→   505→    recommendedQty: qty, status, reason,
   506→   506→    gap, eoq, moq, maxStock, constraintsApplied,
   507→   507→    totalNeeded, totalSupply, preConstraintQty,
   508→   508→  };
   509→   509→}

   534→   534→    return baseDemand * categoryAdj[season];

   539→   539→export function getSeasonForDate(targetDate: Date): BDSeason {

   542→   542→  if (month >= 3 && month <= 5) return 'summer';
   543→   543→  if (month >= 6 && month <= 9) return 'monsoon';
   544→   544→  return 'pre_winter';

   628→   628→/**
   629→   629→ * THE CORE ALGORITHM — Full 9-step implementation
   630→   630→ * Calculates when, what, and how much to order for a single SKU.
   631→   631→ */
   632→   632→export function calculateOrderTrigger(input: OrderTriggerInput): OrderTriggerResult {
   633→   633→  const {
   634→   634→    productId, productSku, productName, category,
   635→   635→    currentStock, reservedStock, safetyStock, maxStock = 500,
   636→   636→    reorderPoint, avgDailyDemand,
   637→   637→    forecastedDemand = 0, qtyOnOrder = 0,
   638→   638→    eoq = 100, moq = 50,
   639→   639→    serviceLevel = 0.95,
   640→   640→    bufferDays = 7,
   641→   641→    monsoonAdjustment = 0,
   642→   642→  } = input;
   643→   643→
   644→   644→  const availableStock = currentStock - reservedStock;
   645→   645→  const today = new Date();
   646→   646→  today.setHours(0, 0, 0, 0);
   647→   647→
   648→   648→  // Merge lead time config
   649→   649→  const leadTimeConfig: LeadTimeConfig = { ...DEFAULT_LEAD_TIME, ...(input.leadTimeConfig || {}) };
   650→   650→  if (input.shippingMethod) leadTimeConfig.shippingMethod = input.shippingMethod;
   651→   651→
   652→   652→  // ── STEP 1: Determine shipment and customs durations ──
   653→   653→  const shipmentDays = leadTimeConfig.shippingMethod === 'sea'
   654→   654→    ? leadTimeConfig.seaTransitDays + monsoonAdjustment
   655→   655→    : leadTimeConfig.airTransitDays;
   656→   656→  const customsDays = leadTimeConfig.shippingMethod === 'sea'
   657→   657→    ? leadTimeConfig.customsClearanceDays
   658→   658→    : 3;
   659→   659→
   660→   660→  // ── STEP 2: Calculate Reorder Point (Q1) ──
   661→   661→  const totalLeadTime = leadTimeConfig.manufacturingDays + shipmentDays + customsDays;
   662→   662→  const effectiveReorderPoint = reorderPoint > 0
   663→   663→    ? reorderPoint
   664→   664→    : (avgDailyDemand * totalLeadTime) + safetyStock;
   665→   665→
   666→   666→  // Determine stock status
   667→   667→  let stockStatus: StockStatus;
   668→   668→  let needsOrder: boolean;
   669→   669→
   670→   670→  if (availableStock <= 0) {
   671→   671→    stockStatus = 'stockout';
   672→   672→    needsOrder = true;
   673→   673→  } else if (availableStock <= safetyStock) {
   674→   674→    stockStatus = 'at_safety';
   675→   675→    needsOrder = true;
   676→   676→  } else if (availableStock <= effectiveReorderPoint) {
   677→   677→    stockStatus = 'below_reorder';
   678→   678→    needsOrder = true;
   679→   679→  } else if (availableStock > maxStock) {
   680→   680→    stockStatus = 'overstock';
   681→   681→    needsOrder = false;
   682→   682→  } else {
   683→   683→    stockStatus = 'adequate';
   684→   684→    needsOrder = availableStock < effectiveReorderPoint;
   685→   685→  }
   686→   686→
   687→   687→  const daysOfStock = avgDailyDemand > 0
   688→   688→    ? Math.floor(availableStock / avgDailyDemand)
   689→   689→    : 999;
   690→   690→
   691→   691→  // ── STEP 3: Calculate Order Trigger Date (Q3) ──
   692→   692→  let daysUntilSafetyStock: number;
   693→   693→  if (avgDailyDemand > 0) {
   694→   694→    daysUntilSafetyStock = (availableStock - safetyStock) / avgDailyDemand;
   695→   695→  } else {
   696→   696→    daysUntilSafetyStock = Infinity;
   697→   697→  }
   698→   698→
   699→   699→  const safetyStockHitDate = addDays(today, Math.max(0, Math.round(daysUntilSafetyStock)));
   700→   700→  let orderTriggerDate = addDays(safetyStockHitDate, -(totalLeadTime + bufferDays));
   701→   701→
   702→   702→  // ── STEP 4: CNY Risk Assessment ──
   703→   703→  const cnyInfo = getCNYForDate(today);
   704→   704→  let cnyRisk = assessCNYRisk(orderTriggerDate, leadTimeConfig.manufacturingDays, cnyInfo, today);
   705→   705→
   706→   706→  // If CNY risk detected, revise order trigger date
   707→   707→  let cnyDelayDays = 0;
   708→   708→  if (cnyRisk.hasRisk) {
   709→   709→    if (cnyRisk.strategy === 'before_cny' && cnyRisk.latestSafeOrderDate && cnyRisk.latestSafeOrderDate >= today) {
   710→   710→      // Order early enough to finish before CNY
   711→   711→      orderTriggerDate = new Date(Math.min(orderTriggerDate.getTime(), cnyRisk.latestSafeOrderDate.getTime()));
   712→   712→    } else if (cnyRisk.strategy === 'after_cny' && cnyRisk.postCnyOrderDate) {
   713→   713→      // Accept delay, order after CNY
   714→   714→      orderTriggerDate = cnyRisk.postCnyOrderDate;
   715→   715→      cnyDelayDays = cnyRisk.additionalDelayDays;
   716→   716→    }
   717→   717→  }
   718→   718→
   719→   719→  // ── STEP 5: Build Complete Timeline ──
   720→   720→  const timeline = buildTimeline(
   721→   721→    orderTriggerDate,
   722→   722→    leadTimeConfig.manufacturingDays,
   723→   723→    shipmentDays,
   724→   724→    customsDays,
   725→   725→    leadTimeConfig.orderProcessingDays,
   726→   726→    leadTimeConfig.packingLoadingDays,
   727→   727→    leadTimeConfig.warehouseTransportDays,
   728→   728→    leadTimeConfig.qcShelvingDays,
   729→   729→    cnyDelayDays,
   730→   730→  );
   731→   731→
   732→   732→  // ── STEP 6: Determine Urgency ──
   733→   733→  const daysUntilTrigger = Math.round((orderTriggerDate.getTime() - today.getTime()) / (1000 * 60 * 60 * 24));
   734→   734→
   735→   735→  let urgency: Urgency;
   736→   736→  if (daysUntilTrigger <= 0) urgency = 'critical';
   737→   737→  else if (daysUntilTrigger <= 30) urgency = 'critical';
   738→   738→  else if (daysUntilTrigger <= 90) urgency = 'high';
   739→   739→  else if (daysUntilTrigger <= 180) urgency = 'normal';
   740→   740→  else urgency = 'low';
   741→   741→
   742→   742→  // Upgrade urgency if CNY causes stockout gap
   743→   743→  if (cnyRisk.hasRisk && urgency === 'normal' && cnyRisk.strategy === 'after_cny') {
   744→   744→    const availableDate = timeline.availableForSaleDate;
   745→   745→    if (safetyStockHitDate < availableDate) {
   746→   746→      const stockoutDays = Math.round((availableDate.getTime() - safetyStockHitDate.getTime()) / (1000 * 60 * 60 * 24));
   747→   747→      if (stockoutDays > 30) urgency = 'high';
   748→   748→    }
   749→   749→  }
   750→   750→
   751→   751→  // ── STEP 7: Calculate Recommended Order Quantity (Q2) ──
   752→   752→  const qtyBreakdown = calculateRecommendedQty(
   753→   753→    forecastedDemand,
   754→   754→    safetyStock,
   755→   755→    availableStock,
   756→   756→    qtyOnOrder,
   757→   757→    eoq,
   758→   758→    moq,
   759→   759→    maxStock,
   760→   760→  );
   761→   761→
   762→   762→  // ── STEP 8: Recommend shipment mode ──
   763→   763→  let recommendedShipmentMode: ShipmentMode = leadTimeConfig.shippingMethod;
   764→   764→  if (urgency === 'critical' && leadTimeConfig.shippingMethod === 'sea' && daysUntilTrigger <= 0) {
   765→   765→    recommendedShipmentMode = 'air';
   766→   766→  }
   767→   767→
   768→   768→  // ── STEP 9: Return complete result ──
   769→   769→  const currentSeason = getBDSeason(today.getMonth() + 1);
   770→   770→  const seasonNote = getSeasonNote(currentSeason.season);
   771→   771→
   772→   772→  const leadTime = calculateTotalLeadTime(leadTimeConfig);
   773→   773→
   774→   774→  return {
   775→   775→    productId, productSku, productName,
   776→   776→    needsOrder, reorderPoint: Math.round(effectiveReorderPoint),
   777→   777→    currentStock, availableStock, safetyStock, stockStatus, daysOfStock,
   778→   778→    suggestedOrderQty: qtyBreakdown.recommendedQty,
   779→   779→    qtyBreakdown,
   780→   780→    orderTriggerDate, reorderHitDate: safetyStockHitDate,
   781→   781→    expectedDeliveryDate: timeline.availableForSaleDate,
   782→   782→    timeline, urgency, daysUntilTrigger,
   783→   783→    totalLeadTimeDays: timeline.totalLeadTimeDays,
   784→   784→    leadTimeBreakdown: leadTime.breakdown,
   785→   785→    cnyRisk,
   786→   786→    recommendedShipmentMode,
   787→   787→    currentSeason: currentSeason.season,
   788→   788→    seasonNote,
   789→   789→  };
   790→   790→}
   791→   791→

   856→   856→  inputs: OrderTriggerInput[]

   859→   859→    .map(i => safeCalculateOrderTrigger(i))

   862→   862→      return priorityOrder[a.urgency] - priorityOrder[b.urgency];
   863→   863→    });
   864→   864→}

   914→   914→export function runOrderTriggerPipeline(
   915→   915→  tenantId: string,
   916→   916→  products: OrderTriggerInput[],
   917→   917→  targetSeason: BDSeason,
   918→   918→  targetYear: number,
   919→   919→  unitCosts: Record<string, number> = {},
   920→   920→  topN: number = 50,
   921→   921→): PipelineResult {
   922→   922→  const today = new Date();
   923→   923→  today.setHours(0, 0, 0, 0);
   924→   924→
   925→   925→  const seasonRange = getSeasonDateRange(targetSeason, targetYear);
   926→   926→  const cnyInfo = getCNYForDate(today);
   927→   927→
   928→   928→  // Calculate triggers for all products
   929→   929→  const triggers = calculateBatchOrderTriggers(products);
   930→   930→
   931→   931→  // Build pipeline products with seasonal adjustments
   932→   932→  const pipelineProducts: PipelineProduct[] = triggers
   933→   933→    .filter(t => t.needsOrder)
   934→   934→    .map(t => {
   935→   935→      const seasonalWeight = t.currentSeason
   936→   936→        ? applySeasonalWeight(1.0, t.currentSeason, t.currentSeason)
   937→   937→        : 1.0;
   938→   938→      const baseDemand = t.suggestedOrderQty;
   939→   939→      const adjustedDemand = Math.round(baseDemand * seasonalWeight);
   940→   940→      const unitCost = unitCosts[t.productSku] || 0;
   941→   941→
   942→   942→      return {
   943→   943→        productSku: t.productSku,
   944→   944→        productName: t.productName,
   945→   945→        category: products.find(p => p.productSku === t.productSku)?.category,
   946→   946→        forecastedDemand: baseDemand,
   947→   947→        adjustedDemand,
   948→   948→        recommendedQty: t.suggestedOrderQty,
   949→   949→        orderTriggerDate: formatDate(t.orderTriggerDate),
   950→   950→        expectedAvailableDate: formatDate(t.expectedDeliveryDate),
   951→   951→        urgency: t.urgency,
   952→   952→        daysUntilTrigger: t.daysUntilTrigger,
   953→   953→        cnyRisk: t.cnyRisk.hasRisk,
   954→   954→        cnyStrategy: t.cnyRisk.strategy,
   955→   955→        totalLeadTimeDays: t.totalLeadTimeDays,
   956→   956→        unitCostBdt: unitCost,
   957→   957→        totalCostBdt: t.suggestedOrderQty * unitCost,
   958→   958→      };
   959→   959→    });
   960→   960→
   961→   961→  // Sort by urgency, then by adjusted demand
   962→   962→  const urgencyOrder: Record<string, number> = { critical: 0, high: 1, normal: 2, low: 3 };
   963→   963→  pipelineProducts.sort((a, b) => {
   964→   964→    const uDiff = (urgencyOrder[a.urgency] || 99) - (urgencyOrder[b.urgency] || 99);
   965→   965→    if (uDiff !== 0) return uDiff;
   966→   966→    return b.adjustedDemand - a.adjustedDemand;
   967→   967→  });
   968→   968→
   969→   969→  const topProducts = pipelineProducts.slice(0, topN);
   970→   970→
   971→   971→  // Summary
   972→   972→  const urgencyCounts = { critical: 0, high: 0, normal: 0, low: 0 };
   973→   973→  topProducts.forEach(p => { urgencyCounts[p.urgency]++; });
   974→   974→
   975→   975→  const orderDates = topProducts.filter(p => p.recommendedQty > 0).map(p => p.orderTriggerDate);
   976→   976→  const totalSpend = topProducts.reduce((sum, p) => sum + p.totalCostBdt, 0);
   977→   977→
   978→   978→  return {
   979→   979→    forecastSessionId: `fs_${formatDateCompact(today)}_${targetSeason}${targetYear}`,
   980→   980→    tenantId,
   981→   981→    season: `${targetSeason}_${targetYear}`,
   982→   982→    period: {
   983→   983→      start: formatDate(seasonRange.start),
   984→   984→      end: formatDate(seasonRange.end),
   985→   985→      totalDays: Math.round((seasonRange.end.getTime() - seasonRange.start.getTime()) / (1000 * 60 * 60 * 24)),
   986→   986→    },
   987→   987→    generatedAt: today.toISOString(),
   988→   988→    cnyWindow: cnyInfo ? {
   989→   989→      year: cnyInfo.year,
   990→   990→      shutdownStart: formatDate(cnyInfo.startDate),
   991→   991→      shutdownEnd: formatDate(cnyInfo.endDate),
   992→   992→    } : null,
   993→   993→    products: topProducts,
   994→   994→    summary: {
   995→   995→      totalProducts: topProducts.length,
   996→   996→      totalRecommendedUnits: topProducts.reduce((sum, p) => sum + p.recommendedQty, 0),
   997→   997→      totalRecommendedSpendBdt: totalSpend,
   998→   998→      criticalUrgencyCount: urgencyCounts.critical,
   999→   999→      highUrgencyCount: urgencyCounts.high,
  1000→  1000→      normalUrgencyCount: urgencyCounts.normal,
  1001→  1001→      lowUrgencyCount: urgencyCounts.low,
  1002→  1002→      cnyRiskCount: topProducts.filter(p => p.cnyRisk).length,
  1003→  1003→      earliestOrderDate: orderDates.length > 0 ? orderDates.sort()[0] : null,
  1004→  1004→      latestOrderDate: orderDates.length > 0 ? orderDates.sort().reverse()[0] : null,
  1005→  1005→    },
  1006→  1006→  };
  1007→  1007→}

  1020→  1020→export function calculateStockProjection(
  1021→  1021→  currentStock: number,
  1022→  1022→  avgDailyDemand: number,
  1023→  1023→  safetyStock: number,
  1024→  1024→  reorderPoint: number,
  1025→  1025→  orderArrivalDate: Date,
  1026→  1026→  orderQty: number,
  1027→  1027→  horizonDays: number = 180,
  1028→  1028→): StockProjectionPoint[] {
  1029→  1029→  const points: StockProjectionPoint[] = [];
  1030→  1030→  const today = new Date();
  1031→  1031→  today.setHours(0, 0, 0, 0);
  1032→  1032→
  1033→  1033→  let stock = currentStock;
  1034→  1034→  let hasArrived = false;
  1035→  1035→
  1036→  1036→  for (let d = 0; d <= horizonDays; d++) {
  1037→  1037→    const date = addDays(today, d);
  1038→  1038→
  1039→  1039→    // Check if order arrives today
  1040→  1040→    if (!hasArrived && date >= orderArrivalDate) {
  1041→  1041→      stock += orderQty;
  1042→  1042→      hasArrived = true;
  1043→  1043→      points.push({ date: formatDate(date), stock: Math.round(stock), event: 'arrival', note: `Order arrives: +${orderQty} units` });
  1044→  1044→      continue;
  1045→  1045→    }
  1046→  1046→
  1047→  1047→    // Daily consumption
  1048→  1048→    stock -= avgDailyDemand;
  1049→  1049→
  1050→  1050→    let event: StockProjectionPoint['event'];
  1051→  1051→    let note: string | undefined;
  1052→  1052→
  1053→  1053→    if (stock <= 0 && stock + avgDailyDemand > 0) {
  1054→  1054→      event = 'stockout'; note = 'Stockout!';
  1055→  1055→    } else if (stock <= safetyStock && stock + avgDailyDemand > safetyStock) {
  1056→  1056→      event = 'safety_hit'; note = 'Stock hits safety level';
  1057→  1057→    } else if (stock <= reorderPoint && stock + avgDailyDemand > reorderPoint) {
  1058→  1058→      event = 'reorder_hit'; note = 'Stock hits reorder point';
  1059→  1059→    }
  1060→  1060→
  1061→  1061→    points.push({ date: formatDate(date), stock: Math.round(Math.max(0, stock)), event, note });
  1062→  1062→  }
  1063→  1063→
  1064→  1064→  return points;
  1065→  1065→}
