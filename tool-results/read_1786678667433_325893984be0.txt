    49→function addDays(iso: string, days: number): Date {
    50→  const d = new Date(iso);
    51→  d.setDate(d.getDate() + days);
    52→  return d;
    53→}
    54→
    55→function formatDate(date: Date): string {
    56→  return date.toLocaleDateString('en-US', {
    57→    month: 'short',
    58→    day: 'numeric',
    59→  });
    60→}
    61→
    62→function formatDateFull(date: Date): string {
    63→  return date.toLocaleDateString('en-US', {
    64→    month: 'short',
    65→    day: 'numeric',
    66→    year: 'numeric',
    67→  });
    68→}
    69→
    70→function daysBetween(a: Date, b: Date): number {
    71→  return Math.round((b.getTime() - a.getTime()) / (1000 * 60 * 60 * 24));
    72→}

   128→function CompactTimeline({
   129→  milestones,
   130→  totalDays,
   131→  todayOffset,
   132→  cnyOffset,
   133→  cnyWidth,
   134→}: {
   135→  milestones: Milestone[];
   136→  totalDays: number;
   137→  todayOffset: number | null;
   138→  cnyOffset: number | null;
   139→  cnyWidth: number | null;
   140→}) {
   141→  return (
   142→    <div className="relative w-full py-1">
   143→      {/* Background track */}
   144→      <div className="flex h-3 rounded-full overflow-hidden bg-gray-100">
   145→        {milestones.slice(0, -1).map((m, i) => {
   146→          const next = milestones[i + 1];
   147→          const segDays = next.dayOffset - m.dayOffset;
   148→          const widthPct = (segDays / totalDays) * 100;
   149→          const colorClass =
   150→            i === 0
   151→              ? SEGMENT_COLORS.manufacturing
   152→              : i === 1
   153→                ? SEGMENT_COLORS.shipping
   154→                : i === 2
   155→                  ? SEGMENT_COLORS.customs
   156→                  : SEGMENT_COLORS.internal;
   157→          return (
   158→            <motion.div
   159→              key={i}
   160→              className={`${colorClass} first:rounded-l-full last:rounded-r-full`}
   161→              style={{ width: `${widthPct}%`, transformOrigin: 'left' }}
   162→              variants={barVariants}
   163→              initial="hidden"
   164→              animate="visible"
   165→            />
   166→          );
   167→        })}
   168→      </div>
   169→
   170→      {/* CNY overlay */}
   171→      {cnyOffset !== null && cnyWidth !== null && (
   172→        <div
   173→          className="absolute top-1 h-3 bg-red-400/40 rounded-sm"
   174→          style={{
   175→            left: `${(cnyOffset / totalDays) * 100}%`,
   176→            width: `${(cnyWidth / totalDays) * 100}%`,
   177→          }}
   178→        />
   179→      )}
   180→
   181→      {/* Today marker */}
   182→      {todayOffset !== null && todayOffset >= 0 && todayOffset <= totalDays && (
   183→        <div
   184→          className="absolute top-0 bottom-0 w-0.5 bg-slate-800"
   185→          style={{ left: `${(todayOffset / totalDays) * 100}%` }}
   186→        >
   187→          <div className="absolute -top-1 left-1/2 -translate-x-1/2 w-1.5 h-1.5 rounded-full bg-slate-800" />
   188→        </div>
   189→      )}
   190→
   191→      {/* Milestone dots */}
   192→      <div className="absolute inset-0 flex items-center">
   193→        {milestones.map((m, i) => (
   194→          <Tooltip key={i}>
   195→            <TooltipTrigger asChild>
   196→              <motion.div
   197→                className="absolute w-3 h-3 rounded-full border-2 border-white shadow-sm cursor-default"
   198→                style={{
   199→                  left: `${(m.dayOffset / totalDays) * 100}%`,
   200→                  transform: 'translateX(-50%)',
   201→                  backgroundColor:
   202→                    i === 0
   203→                      ? '#475569'
   204→                      : i === milestones.length - 1
   205→                        ? '#10b981'
   206→                        : '#3b82f6',
   207→                }}
   208→                variants={itemVariants}
   209→                initial="hidden"
   210→                animate="visible"
   211→              />
   212→            </TooltipTrigger>
   213→            <TooltipContent side="top">
   214→              <p className="font-medium">{m.label}</p>
   215→              <p className="text-[10px] opacity-80">{formatDateFull(m.date)}</p>
   216→            </TooltipContent>
   217→          </Tooltip>
   218→        ))}
   219→      </div>
   220→    </div>
   221→  );
   222→}

   566→
   567→export function OrderTimeline({
   568→  orderDate,
   569→  leadTimeBreakdown,
   570→  shippingMethod,
   571→  cnyRisk = false,
   572→  cnyDelayDays = 0,
   573→  cnyPeriod,
   574→  compact = false,
   575→}: OrderTimelineProps) {
   576→  const { milestones, segments, totalDays, todayOffset, cnyOffset, cnyWidth } = useMemo(() => {
   577→    const bd = leadTimeBreakdown;
   578→    const total = bd.manufacturing + bd.shipping + bd.customs + bd.internal;
   579→
   580→    // Cumulative offsets for each milestone
   581→    const offsets = [
   582→      0,
   583→      bd.manufacturing,
   584→      bd.manufacturing + bd.shipping,
   585→      bd.manufacturing + bd.shipping + bd.customs,
   586→      bd.manufacturing + bd.shipping + bd.customs + bd.internal,
   587→    ];
   588→    // The 6th milestone is same as 5th (Available = after internal processing)
   589→    // But per spec: 6 milestones from Order to Availability
   590→    // Offsets: 0, mfg, mfg+ship, mfg+ship+customs, mfg+ship+customs+internal
   591→    // But the 6th milestone "Available" is at the same offset as the 5th "Customs Cleared" + internal
   592→
   593→    const milestoneDates = offsets.map((off) => addDays(orderDate, off));
   594→
   595→    const milestoneList: Milestone[] = [
   596→      {
   597→        label: 'Order Placed',
   598→        icon: <Package className="h-3.5 w-3.5" />,
   599→        date: milestoneDates[0],
   600→        dayOffset: offsets[0],
   601→        color: 'slate',
   602→      },
   603→      {
   604→        label: 'Mfg Complete',
   605→        icon: <Factory className="h-3.5 w-3.5" />,

   613→        date: milestoneDates[2],
   614→        dayOffset: offsets[2],
   615→        color: 'emerald',

   617→      {

   640→    // Fix: "Arrived at Port" is at the end of shipping, which is offsets[2]
   641→    // But shipping includes transit, so "Shipped" is start of shipping, "Arrived" is end of shipping
   642→    // Let me recalculate: Shipped = after manufacturing starts shipping (offsets[1])
   643→    // Arrived at Port = after shipping completes (offsets[2])
   644→    milestoneList[2] = {
   645→      ...milestoneList[2],
   646→      date: addDays(orderDate, offsets[1]), // Shipped date = after manufacturing
   647→      dayOffset: offsets[1],
   648→    };
   649→    milestoneList[3] = {

   651→      date: addDays(orderDate, offsets[2]), // Arrived at Port = after shipping

   653→    };

   687→    const todayOff = daysBetween(orderD, today);
   688→
   689→    // CNY overlay calculation
   690→    let cnyOff: number | null = null;
   691→    let cnyW: number | null = null;
   692→    if (cnyPeriod) {

   734→          </p>
   735→        </TooltipContent>
   736→      </Tooltip>
   737→    );
   738→  }
   739→
   740→  return (
   741→    <FullTimeline
   742→      milestones={milestones}
   743→      segments={segments}
   744→      totalDays={totalDays}
   745→      todayOffset={todayOffset}
   746→      cnyOffset={cnyOffset}
   747→      cnyWidth={cnyWidth}
   748→      orderDate={orderDate}
   749→      cnyRisk={cnyRisk}
   750→      cnyDelayDays={cnyDelayDays}
   751→      cnyPeriod={cnyPeriod}
   752→      shippingMethod={shippingMethod}
   753→    />
   754→  );
