    85→function ValidateStep() {
    86→  const { currentImportId, setLoading, setError, setValidationResult, setStep, isLoading } = useETLStore();
    87→  const handleProceed = useCallback(() => { setStep(3); }, [setStep]);
    88→
    89→  useEffect(() => {
    90→    const runValidation = async () => {
    91→      if (!currentImportId) return;
    92→      setLoading(true);
    93→      setError(null);
    94→      try {
    95→        const res = await fetch(`/api/imports/${currentImportId}/validate`, { method: 'POST' });
    96→        if (!res.ok) { const err = await res.json().catch(() => ({})); throw new Error(err.error || 'Validation failed'); }
    97→        const json = await res.json();
    98→        if (!json.success) throw new Error(json.error || 'Validation failed');
    99→        setValidationResult(json.data as ValidationResultClient);
   100→      } catch (err) { setError(err instanceof Error ? err.message : 'Validation failed'); }
   101→      finally { setLoading(false); }
   102→    };
   103→    runValidation();
   104→  }, [currentImportId, setLoading, setError, setValidationResult]);
   105→
   106→  return (
   107→    <div>
   108→      {isLoading && (
   109→        <Card className="mb-4">
   110→          <CardContent className="py-8 text-center">
   111→            <Loader2 className="h-8 w-8 mx-auto text-blue-500 animate-spin mb-3" />
   112→            <p className="text-sm text-gray-600">Running 3-phase validation...</p>
   113→            <p className="text-xs text-gray-400 mt-1">Schema → Data → Business Rules</p>
   114→          </CardContent>
   115→        </Card>
   116→      )}
   117→      <ValidationResults onProceed={handleProceed} />
   118→    </div>
   119→  );
   120→}

   122→function HarmonizeStep() {
   123→  const { currentImportId, setLoading, setError, setHarmonizationResult, setStep, harmonizationResult, isLoading } = useETLStore();
   124→  const handleProceed = useCallback(() => { setStep(4); }, [setStep]);
   125→
   126→  useEffect(() => {
   127→    if (harmonizationResult) return;
   128→    const runHarmonization = async () => {
   129→      if (!currentImportId) return;
   130→      setLoading(true); setError(null);
   131→      try {
   132→        const res = await fetch(`/api/imports/${currentImportId}/harmonize`, { method: 'POST' });
   133→        if (!res.ok) { const err = await res.json().catch(() => ({})); throw new Error(err.error || 'Harmonization failed'); }
   134→        const json = await res.json();
   135→        if (!json.success) throw new Error(json.error || 'Harmonization failed');
   136→        setHarmonizationResult(json.data as HarmonizationResultClient);
   137→      } catch (err) { setError(err instanceof Error ? err.message : 'Harmonization failed'); }
   138→      finally { setLoading(false); }
   139→    };
   140→    runHarmonization();
   141→  }, [currentImportId, harmonizationResult, setLoading, setError, setHarmonizationResult]);
   142→
   143→  if (isLoading && !harmonizationResult) {
   144→    return (
   145→      <Card>
   146→        <CardContent className="py-8 text-center">
   147→          <Loader2 className="h-8 w-8 mx-auto text-blue-500 animate-spin mb-3" />
   148→          <p className="text-sm text-gray-600">Running 6-step harmonization...</p>
   149→          <p className="text-xs text-gray-400 mt-1">Trim → Dates → Categories → Units → Dedup → Enrich</p>
   150→        </CardContent>
   151→      </Card>
   152→    );
   153→  }
   154→  return <HarmonizationLog onProceed={handleProceed} />;
   155→}

   157→function InsertStep() {
   158→  const { currentImportId, setLoading, setError, setInsertionResult, setStep, insertionResult, uploadResult, mappings, importType, isLoading } = useETLStore();
   159→
   160→  useEffect(() => {
   161→    if (insertionResult) return;
   162→    const runInsert = async () => {
   163→      if (!currentImportId) return;
   164→      setLoading(true); setError(null);
   165→      try {
   166→        const res = await fetch(`/api/imports/${currentImportId}/insert`, { method: 'POST' });
   167→        if (!res.ok) { const err = await res.json().catch(() => ({})); throw new Error(err.error || 'Insertion failed'); }
   168→        const json = await res.json();
   169→        if (!json.success) throw new Error(json.error || 'Insertion failed');
   170→        setInsertionResult(json.data as InsertionResultClient);
   171→      } catch (err) { setError(err instanceof Error ? err.message : 'Insertion failed'); }
   172→      finally { setLoading(false); }
   173→    };
   174→    runInsert();
   175→  }, [currentImportId, insertionResult, setLoading, setError, setInsertionResult]);
   176→
   177→  if (isLoading && !insertionResult) {
   178→    return (
   179→      <Card>
   180→        <CardContent className="py-8 text-center">
   181→          <Loader2 className="h-8 w-8 mx-auto text-blue-500 animate-spin mb-3" />
   182→          <p className="text-sm text-gray-600">Inserting data into database...</p>
   183→        </CardContent>
   184→      </Card>
   185→    );
   186→  }
   187→  if (!insertionResult) return null;
   188→
   189→  const schema = importType ? IMPORT_TYPE_SCHEMAS[importType as ImportType] : null;
   190→  const requiredTotal = schema?.requiredFields.length || 0;
   191→  const requiredMapped = mappings.filter((m) => m.targetField && m.confidence > 0 && schema?.requiredFields.some((rf) => rf.field === m.targetField)).length;
   192→  const qualityStats: QualityStats = {
   193→    rowsTotal: uploadResult?.rowsTotal || 0,
   194→    rowsValid: uploadResult?.rowsTotal || insertionResult.inserted,
   195→    rowsInserted: insertionResult.inserted,
   196→    rowsDuplicate: 0,
   197→    requiredMapped,
   198→    requiredTotal,
   199→  };
   200→
   201→  return (
   202→    <motion.div initial={{ opacity: 0, y: 20 }} animate={{ opacity: 1, y: 0 }} transition={{ duration: 0.4 }} className="space-y-4">
   203→      <Card className="border-emerald-200 bg-emerald-50/50">
   204→        <CardContent className="py-6">
   205→          <div className="flex items-center gap-3 mb-4">
   206→            <div className="w-12 h-12 rounded-full bg-emerald-100 flex items-center justify-center">
   207→              <CheckCircle2 className="h-7 w-7 text-emerald-600" />
   208→            </div>
   209→            <div>
   210→              <h2 className="text-xl font-bold text-emerald-800">Import Complete!</h2>
   211→              <p className="text-sm text-emerald-600">Your data has been successfully processed and inserted</p>
   212→            </div>
   213→          </div>
   214→          <QualityBadge score={insertionResult.qualityScore} size="lg" showBreakdown={true} stats={qualityStats} />
   215→        </CardContent>
   216→      </Card>
   217→      <div className="grid grid-cols-2 sm:grid-cols-4 gap-3">
   218→        <Card><CardContent className="pt-4 pb-3 text-center"><Database className="h-5 w-5 mx-auto text-blue-500 mb-1" /><div className="text-2xl font-bold text-gray-700">{insertionResult.inserted}</div><div className="text-xs text-gray-500">Rows Inserted</div></CardContent></Card>
   219→        <Card><CardContent className="pt-4 pb-3 text-center"><XCircle className="h-5 w-5 mx-auto text-red-500 mb-1" /><div className="text-2xl font-bold text-gray-700">{insertionResult.skipped}</div><div className="text-xs text-gray-500">Rows Skipped</div></CardContent></Card>
   220→        <Card><CardContent className="pt-4 pb-3 text-center"><FileSpreadsheet className="h-5 w-5 mx-auto text-gray-500 mb-1" /><div className="text-2xl font-bold text-gray-700">{insertionResult.createdMasterData?.length || 0}</div><div className="text-xs text-gray-500">Master Data Created</div></CardContent></Card>
   221→        <Card><CardContent className="pt-4 pb-3 text-center"><Loader2 className="h-5 w-5 mx-auto text-gray-400 mb-1" style={{ animation: 'none' }} /><div className="text-2xl font-bold text-gray-700">{insertionResult.durationMs !== null ? insertionResult.durationMs < 1000 ? `${insertionResult.durationMs}ms` : `${(insertionResult.durationMs / 1000).toFixed(1)}s` : '—'}</div><div className="text-xs text-gray-500">Duration</div></CardContent></Card>
   222→      </div>
   223→      {insertionResult.createdMasterData && insertionResult.createdMasterData.length > 0 && (
   224→        <Card>
   225→          <CardHeader className="pb-2"><CardTitle className="text-sm font-medium text-gray-600">Auto-Created Master Data</CardTitle></CardHeader>
   226→          <CardContent><ul className="text-xs text-gray-500 space-y-1">{insertionResult.createdMasterData.map((item, i) => (<li key={i} className="flex items-center gap-2"><Badge variant="outline" className="text-xs h-5">New</Badge>{item}</li>))}</ul></CardContent>
   227→        </Card>
   228→      )}
   229→      {insertionResult.errors && insertionResult.errors.length > 0 && (
   230→        <Card>
   231→          <CardHeader className="pb-2"><CardTitle className="text-sm font-medium text-red-600">Insertion Errors ({insertionResult.errors.length})</CardTitle></CardHeader>
   232→          <CardContent><ul className="text-xs text-red-500 space-y-1">{insertionResult.errors.slice(0, 10).map((err, i) => (<li key={i}>Row {err.row}: {err.error}</li>))}{insertionResult.errors.length > 10 && <li className="text-gray-400">...and {insertionResult.errors.length - 10} more</li>}</ul></CardContent>
   233→        </Card>
   234→      )}
   235→    </motion.div>
   236→  );
   237→}
